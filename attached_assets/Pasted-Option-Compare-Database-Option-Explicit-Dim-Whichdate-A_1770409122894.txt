Private Sub Form_AfterInsert()
If Me.Caption = "Master Task List" Then
Dim Q As Integer, Yoda As DAO.Database, rs As DAO.Recordset, rsj As DAO.Recordset, Ponum As Single, ref As Single
Set Yoda = CurrentDb
Set rs = Yoda.OpenRecordset("Select * From [Task Master]", dbOpenDynaset)
Set rsj = Yoda.OpenRecordset("Select * From [Job Details]", dbOpenSnapshot)

Q = MsgBox("Would you like to insert this task into all jobs", vbYesNo, "Add to all jobs?")
If Q = vbYes Then
    rsj.MoveFirst
    Do Until rsj.EOF
        'CurrentDb.Execute "Insert into [Task Master] ([Job #],[Task #],[Stage],[FinList#],[task],[Supplier/Trade],[WhosTask],[PoRefNum],[Ref],[Memo])" _
        '& " VALUES (" & RsJ![Job #] & "," & Me.TaskNum & ",'" & Me.Stage & "'," & Me.FinList & ",'" & Me.Task & "','" & Me.[Supplier/Trade] _
        '& "','" & Me.WhosTask & "'," & Me.PORefNum & "," & Me.Ref & ",'" & Me.Memo & "');"
        With rs
        .AddNew
        ![Job #] = rsj![Job #]
        ![Task #] = Me.TaskNum
        ![Stage] = Me.Stage
        ![Super] = Me.Super
        ![FinList#] = Me.FinList
        ![Task] = Me.Task
        ![Supplier/Trade] = Me.Supplier_Trade
        ![WhosTask] = Me.WhosTask
        !PORefNum = Me.PORefNum
        !ref = Me.ref
        !Memo = Me.Memo
        .Update
        End With
        rsj.MoveNext
    Loop
End If
rs.Close
rsj.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End If
Newtask = False
End Sub

Private Sub PORefNum_AfterUpdate()
If Newtask = False Then
    Dim Q As Integer, Q2 As Integer, RIS As Integer, RIF As Integer, Rss As Integer, RSF As Integer, RCS As Integer, RCF As Integer
If Me.Caption = "Master Task List" Then
    Q = MsgBox("Do you want to change all jobs in the Task Bible", vbYesNo, "Change all Jobs?")
        If Q = vbYes Then
        CurrentDb.Execute "Update [Task Master] Set [Task Master]![PORefNum]='" & Me.PORefNum & "' Where [Task Master]!Task='" & Me.Task & "'"
        Else
            Q2 = MsgBox("Do you want to select a range of jobs to change in the Task Bible", vbYesNo, "Change some Jobs?")
                If Q2 = vbYes Then
                    RCS = InputBox("Cascades Start", "Enter a starting Cascades-Ironwood range", 0)
                    RCF = InputBox("Cascdes End", "Enter an end to Cascades-Ironwood range", 0)
                    RIS = InputBox("Ironwood Start", "Enter a starting Ironwood range", 0)
                    RIF = InputBox("Ironwood End", "Enter an end to Ironwood range", 0)
                    Rss = InputBox("Salvi Start", "Enter a starting Salvi range", 0)
                    RSF = InputBox("Salvi End", "Enter an end to Salvi range", 0)
                CurrentDb.Execute "Update [Task Master] Set [Task Master]!PORefNum='" & Me.PORefNum _
                & "' Where [Task Master]!Task='" & Me.Task & "' And [Task Master]![Job #] Between " & RCS & " and " & RCF _
                & " Or [Task Master]!Task='" & Me.Task & "' And [Task Master]![Job #] Between " & RIS & " and " & RIF _
                & " OR [Task Master]!Task='" & Me.Task & "' And [Task Master]![Job #] Between " & Rss & " and " & RSF
                End If
        End If
End If
End If

Public Sub OpenPOREPORT()
Dim YodaPO As DAO.Database, records As String, rs As DAO.Recordset, PoReferenceNum As Single, tbd As TableDef, Estexists As Boolean
    Set YodaPO = CurrentDb
    records = "Select * From PONumbers Where PONum=" & Nz(Me.[PO #], 0)
    Set rs = YodaPO.OpenRecordset(records, dbOpenDynaset)
If rs.EOF Then
    MsgBox "Sorry Yoda cannot find this PO, it was printed before my time", vbOKOnly, "Purchase Order Not Found"
Else
    'Make Sure est table exists
    For Each tbd In YodaPO.TableDefs
    If tbd.Name = rs!Job & "Est" Then Estexists = True
    Next
    If Estexists = True Then
        PoReferenceNum = rs!PO
        DoCmd.OpenForm "Password"
        Forms.Password.PasswordBox.Value = "Yoda"
        DoCmd.OpenForm "EMaster"
        Forms.Emaster.Visible = False
        Forms.Emaster.Job = rs!Job
        Forms.Password.PasswordBox.Value = "Super"
        Forms.Password.Visible = False
        DoCmd.OpenForm "POMaster"
        Forms.POMaster.Visible = False
       Forms.POMaster.Job.Value = rs!Job
        Forms.POMaster.PO.Value = PoReferenceNum
        Forms!POMaster.PO_AfterUpdate
        Forms.POMaster.Hide_Click
        TaskMasterPo
        Forms!POMaster.Ponum = Me.PO__
        Forms!POMaster.Recalc
        DoCmd.OpenReport "POReport", acViewPreview
        DoCmd.RunCommand acCmdZoom100
        DoCmd.MoveSize 0, 0, 13000, 9400
        Forms.Password.PasswordBox.Value = ""
        DoCmd.Close acForm, "POMaster"
        DoCmd.Close acForm, "Emaster"
    Else 'Attempt to relink tables
        On Error GoTo Err_LinkEstErr
        Dim Tablename As String, Strloc As String, Newname As String, Opt As Integer
        'First check backend for error If it isn't on the backend we can trap the error
        Tablename = rs!Job & "Est"
        Newname = rs!Job & "EstTemp"

        Strloc = "\\SALVIDC\MSAccess\The_Maestro\BE\Yodabackend.accdb"
        'Has to be imported and then deleted because Link does not cause error if not found
       DoCmd.TransferDatabase acImport, "Microsoft Access", Strloc, acTable, Tablename, Newname
        DoCmd.Rename Newname, acTable, Newname
        DoCmd.TransferDatabase acLink, "Microsoft Access", Strloc, acTable, Tablename, Newname
        DoCmd.DeleteObject acTable, Newname
        DoCmd.Rename Tablename, acTable, Newname & "1"
       
Letsgo:
                Exit Sub
Err_LinkEstErr:
                If err.Number = 3011 Then
                'Check YodaBackendARCHIVE since file does not exist in Backend
                Tablename = Forms.Taskbibleform.JobC.Value & "Est"
                Strloc = "\\SALVIDC\MSAccess\The_Maestro\BE\YodaBackendARCHIVE.accdb"
                DoCmd.TransferDatabase acLink, "Microsoft Access", Strloc, acTable, Tablename, Tablename
                    GoTo Letsgo:
            Else
                Opt = MsgBox("The Estimate table could not be found on the BackEnd" & vbCrLf & "Please Notify a Yoda Master", vbCritical, "The Maestro Warns")
           End If
    End If
End If
                   'Free Memory
                    rs.Close
                    YodaPO.Close
                    Set YodaPO = Nothing
                    Set rs = Nothing
End Sub

Public Sub GetJobDetails()
On Error GoTo err
Dim Taskey As DAO.Database
    Set Taskey = CurrentDb
Dim rs As DAO.Recordset
    Set rs = Taskey.OpenRecordset("Job Details")
 
Dim cr As String
cr = Me.JobC
rs.FindFirst "[Job #]=" & cr
If rs.NoMatch Then 'Do Nothing
Else
Me.BuildingPer = rs![Building Permit #]
Me.WhoseJob = rs![Site Super]
End If


Set rs = OpenForSeek("Customer Information")
rs.Index = "PrimaryKey"
rs.Seek "=", Me.JobC
If rs.NoMatch Then 'Do Nothing
Else
Dim LName As String, FName As String
LName = rs![Last Name]
FName = Nz(rs![First Names], "")
Me.Customer = Namey(LName, FName)
End If
            'Free Memory
            Taskey.Close
            Set Taskey = Nothing
            Set rs = Nothing
Exit Sub
err:
MsgBox err.Number & "--" & err.Description, vbOKOnly, "The Maestro Warns"
End Sub

            End If

    Set Taskey = Nothing
