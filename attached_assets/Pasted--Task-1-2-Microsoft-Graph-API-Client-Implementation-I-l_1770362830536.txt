/**
 * Microsoft Graph API Client Service
 * All Rights Reserved and Preserved; Capomastro Holdings Ltd.
 */

import { ConfidentialClientApplication, AuthenticationResult } from '@azure/msal-node';
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';
import { promises as fs } from 'fs';
import { join } from 'path';

import {
  MsGraphConfig,
  MsGraphTokenResponse,
  MsGraphUserProfile,
  MsGraphDriveItem,
  MsGraphUploadSession,
  MsGraphEmailMessage,
  MsGraphCalendarEvent,
  WopiCheckFileInfo,
  MsGraphError,
  MsGraphErrorCode,
  FileUploadProgress,
  EmailFilter,
  CachedToken
} from '../../shared/types/ms-graph';

// Import Supabase client for token caching
import { SupabaseClient } from '@supabase/supabase-js';
import { Database } from '../../shared/schema';

/**
 * Configuration for Microsoft Graph Client
 */
export interface MsGraphClientConfig {
  auth: MsGraphConfig;
  baseUrl?: string;
  timeout?: number;
  enableCaching?: boolean;
  cacheTtlSeconds?: number;
  maxRetries?: number;
  retryDelayMs?: number;
  userAgent?: string;
  plenumNetSecurityMode?: 'phi' | 'one' | 'zero'; // Integrate with PlenumNET security
}

/**
 * Microsoft Graph API Client
 */
export class MsGraphClient {
  private config: Required<MsGraphClientConfig>;
  private msalClient: ConfidentialClientApplication;
  private axiosInstance: AxiosInstance;
  private tokenCache: Map<string, CachedToken> = new Map();
  
  // Supabase client for persistent token storage
  private supabase?: SupabaseClient<Database>;

  constructor(config: MsGraphClientConfig, supabase?: SupabaseClient<Database>) {
    this.config = {
      baseUrl: 'https://graph.microsoft.com/v1.0',
      timeout: 30000,
      enableCaching: true,
      cacheTtlSeconds: 300,
      maxRetries: 3,
      retryDelayMs: 1000,
      userAgent: 'MaestroERP/1.0',
      plenumNetSecurityMode: 'one',
      ...config
    };

    this.supabase = supabase;

    // Initialize MSAL client
    this.msalClient = new ConfidentialClientApplication({
      auth: {
        clientId: this.config.auth.clientId,
        clientSecret: this.config.auth.clientSecret,
        authority: this.config.auth.authority || `https://login.microsoftonline.com/${this.config.auth.tenantId}`,
      },
      cache: {
        cachePlugin: this.createCachePlugin()
      }
    });

    // Initialize Axios instance
    this.axiosInstance = axios.create({
      baseURL: this.config.baseUrl,
      timeout: this.config.timeout,
      headers: {
        'User-Agent': this.config.userAgent,
        'Content-Type': 'application/json'
      }
    });

    // Add request interceptor for authentication
    this.axiosInstance.interceptors.request.use(
      async (config) => {
        // Get token for the request
        const token = await this.getAccessToken();
        
        // Add PlenumNET security header if configured
        if (this.config.plenumNetSecurityMode) {
          config.headers['X-PlenumNET-Security-Mode'] = this.config.plenumNetSecurityMode;
        }
        
        // Add authorization header
        config.headers.Authorization = `Bearer ${token}`;
        
        return config;
      },
      (error) => Promise.reject(error)
    );

    // Add response interceptor for error handling and retries
    this.axiosInstance.interceptors.response.use(
      (response) => response,
      async (error) => {
        const originalRequest = error.config;
        
        // Handle 401 Unauthorized (token expired)
        if (error.response?.status === 401 && !originalRequest._retry) {
          originalRequest._retry = true;
          
          try {
            // Try to refresh the token
            await this.refreshAccessToken();
            return this.axiosInstance(originalRequest);
          } catch (refreshError) {
            // If refresh fails, clear cache and throw original error
            this.clearTokenCache();
            throw new MsGraphError(
              'Authentication failed after token refresh',
              MsGraphErrorCode.AUTHENTICATION_FAILED,
              401,
              refreshError
            );
          }
        }
        
        // Handle rate limiting (429 Too Many Requests)
        if (error.response?.status === 429) {
          const retryAfter = error.response.headers['retry-after'] || this.config.retryDelayMs;
          await this.delay(parseInt(retryAfter.toString()) * 1000);
          return this.axiosInstance(originalRequest);
        }
        
        // Convert to MsGraphError
        const graphError = this.normalizeGraphError(error);
        throw graphError;
      }
    );
  }

  /**
   * ============================================================================
   * AUTHENTICATION & TOKEN MANAGEMENT
   * ============================================================================
   */

  /**
   * Get access token for API calls
   */
  async getAccessToken(userId?: string): Promise<string> {
    const cacheKey = userId || 'app-only';
    
    // Check cache first
    if (this.config.enableCaching) {
      const cached = this.tokenCache.get(cacheKey);
      if (cached && cached.expiresAt > new Date()) {
        return cached.accessToken;
      }
      
      // Try to load from Supabase cache
      if (this.supabase && userId) {
        const { data } = await this.supabase
          .from('ms_graph_tokens')
          .select('*')
          .eq('user_id', userId)
          .single();
          
        if (data && new Date(data.expires_at) > new Date()) {
          const token: CachedToken = {
            accessToken: data.access_token,
            refreshToken: data.refresh_token,
            expiresAt: new Date(data.expires_at),
            scopes: data.scopes,
            userId: data.user_id,
            tenantId: data.tenant_id
          };
          
          this.tokenCache.set(cacheKey, token);
          return token.accessToken;
        }
      }
    }
    
    // Acquire new token
    let tokenResult: AuthenticationResult;
    
    if (userId) {
      // User-delegated token (requires previous authentication)
      // This would be called after the user has authenticated via OAuth2
      tokenResult = await this.acquireTokenSilent(userId);
    } else {
      // App-only token (client credentials flow)
      tokenResult = await this.acquireAppOnlyToken();
    }
    
    // Cache the token
    const cachedToken: CachedToken = {
      accessToken: tokenResult.accessToken,
      refreshToken: tokenResult.refreshToken || '',
      expiresAt: new Date(Date.now() + (tokenResult.expiresOn?.getTime() || 0)),
      scopes: tokenResult.scopes || [],
      userId: userId || 'app-only',
      tenantId: this.config.auth.tenantId
    };
    
    this.tokenCache.set(cacheKey, cachedToken);
    
    // Persist to Supabase if user token
    if (this.supabase && userId) {
      await this.supabase
        .from('ms_graph_tokens')
        .upsert({
          user_id: userId,
          access_token: cachedToken.accessToken,
          refresh_token: cachedToken.refreshToken,
          expires_at: cachedToken.expiresAt.toISOString(),
          scopes: cachedToken.scopes,
          tenant_id: cachedToken.tenantId,
          updated_at: new Date().toISOString()
        })
        .eq('user_id', userId);
    }
    
    return tokenResult.accessToken;
  }

  /**
   * Acquire app-only token using client credentials
   */
  private async acquireAppOnlyToken(): Promise<AuthenticationResult> {
    try {
      const result = await this.msalClient.acquireTokenByClientCredential({
        scopes: ['https://graph.microsoft.com/.default'],
        authority: `https://login.microsoftonline.com/${this.config.auth.tenantId}`
      });
      
      if (!result) {
        throw new MsGraphError(
          'Failed to acquire app-only token',
          MsGraphErrorCode.AUTHENTICATION_FAILED
        );
      }
      
      return result;
    } catch (error) {
      throw new MsGraphError(
        `App-only token acquisition failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
        MsGraphErrorCode.AUTHENTICATION_FAILED,
        undefined,
        error
      );
    }
  }

  /**
   * Acquire token silently for a user (requires existing auth)
   */
  private async acquireTokenSilent(userId: string): Promise<AuthenticationResult> {
    const account = await this.msalClient.getTokenCache().getAccountByHomeId(userId);
    
    if (!account) {
      throw new MsGraphError(
        'No cached account found for user',
        MsGraphErrorCode.AUTHENTICATION_FAILED
      );
    }
    
    try {
      const result = await this.msalClient.acquireTokenSilent({
        account,
        scopes: this.config.auth.scopes
      });
      
      return result;
    } catch (error) {
      throw new MsGraphError(
        `Silent token acquisition failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
        MsGraphErrorCode.AUTHENTICATION_FAILED,
        undefined,
        error
      );
    }
  }

  /**
   * Refresh access token
   */
  async refreshAccessToken(userId?: string): Promise<void> {
    const cacheKey = userId || 'app-only';
    this.tokenCache.delete(cacheKey);
    
    if (this.supabase && userId) {
      await this.supabase
        .from('ms_graph_tokens')
        .delete()
        .eq('user_id', userId);
    }
    
    // Force new token acquisition
    await this.getAccessToken(userId);
  }

  /**
   * Clear token cache
   */
  clearTokenCache(): void {
    this.tokenCache.clear();
  }

  /**
   * Get user's emails with filtering
   */
  async getUserEmails(userId: string, filter?: EmailFilter): Promise<any[]> {
    try {
      // Build OData filter query
      const filterParts: string[] = [];
      
      if (filter?.from?.length) {
        filterParts.push(`from/emailAddress/address in ('${filter.from.join("','")}')`);
      }
      
      if (filter?.subjectContains?.length) {
        const subjectFilters = filter.subjectContains.map(subj => 
          `contains(subject, '${subj.replace(/'/g, "''")}')`
        );
        filterParts.push(`(${subjectFilters.join(' or ')})`);
      }
      
      if (filter?.receivedAfter) {
        filterParts.push(`receivedDateTime ge ${filter.receivedAfter.toISOString()}`);
      }
      
      if (filter?.receivedBefore) {
        filterParts.push(`receivedDateTime le ${filter.receivedBefore.toISOString()}`);
      }
      
      if (filter?.hasAttachments !== undefined) {
        filterParts.push(`hasAttachments eq ${filter.hasAttachments}`);
      }
      
      const filterQuery = filterParts.length > 0 ? `$filter=${filterParts.join(' and ')}` : '';
      const queryParams = filterQuery ? `?${filterQuery}&$top=50` : '?$top=50';
      
      const response = await this.axiosInstance.get(
        `/users/${userId}/messages${queryParams}`
      );
      
      return response.data.value;

  /**
   * Normalize Graph API errors
   */
  private normalizeGraphError(error: any): MsGraphError {
    if (error instanceof MsGraphError) {
      return error;
    }
    
    const statusCode = error.response?.status;
    const graphError = error.response?.data?.error;
    
    let code: MsGraphErrorCode;
    let message: string;
    
    if (statusCode === 401) {
      code = MsGraphErrorCode.AUTHENTICATION_FAILED;
      message = 'Authentication failed - invalid or expired token';
    } else if (statusCode === 403) {
      code = MsGraphErrorCode.INSUFFICIENT_PERMISSIONS;
      message = 'Insufficient permissions to perform this action';
    } else if (statusCode === 404) {
      if (error.config?.url?.includes('/drive/items/')) {
        code = MsGraphErrorCode.FILE_NOT_FOUND;
        message = 'File not found in OneDrive';
      } else if (error.config?.url?.includes('/users/')) {
        code = MsGraphErrorCode.USER_NOT_FOUND;
        message = 'User not found';
      } else {
        code = MsGraphErrorCode.NETWORK_ERROR;
        message = 'Resource not found';
      }
    } else if (statusCode === 429) {
      code = MsGraphErrorCode.RATE_LIMITED;
      message = 'Rate limited by Microsoft Graph API';
    } else if (error.code === 'ECONNREFUSED' || error.code === 'ETIMEDOUT') {
      code = MsGraphErrorCode.NETWORK_ERROR;
      message = 'Network error connecting to Microsoft Graph API';
    } else {
      code = MsGraphErrorCode.NETWORK_ERROR;
      message = graphError?.message || error.message || 'Unknown Microsoft Graph API error';
    }
    
    return new MsGraphError(message, code, statusCode, graphError);
  }

  /**
   * Create cache plugin for MSAL
   */
  private createCachePlugin(): any {
    return {
      beforeCacheAccess: async (cacheContext: any) => {
        // Load cache from file or database
        try {
          const cachePath = join(process.cwd(), '.cache', 'msal-cache.json');
          const cacheData = await fs.readFile(cachePath, 'utf-8');
          cacheContext.tokenCache.deserialize(cacheData);
        } catch (error) {
          // Cache file doesn't exist, start with empty cache
        }
      },
      afterCacheAccess: async (cacheContext: any) => {
        // Save cache to file or database
        if (cacheContext.cacheHasChanged) {
          try {
            const cacheDir = join(process.cwd(), '.cache');
            await fs.mkdir(cacheDir, { recursive: true });
            
            const cachePath = join(cacheDir, 'msal-cache.json');
            await fs.writeFile(cachePath, cacheContext.tokenCache.serialize());
          } catch (error) {
            console.error('Failed to save MSAL cache:', error);
          }
        }
      }
    };
  }

  /**
   * Delay helper for retries
   */
  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
