# **COMPLETE TECHNICAL SPECIFICATION & IMPLEMENTATION GUIDE: Modular, Multi-Tenant ERP for Construction**

## **PROJECT OVERVIEW & ARCHITECTURAL PHILOSOPHY**

### **Core Mission**
Build a white-label, SaaS-enabled Enterprise Resource Planning (ERP) system specifically architected for residential construction and land development firms. The system will replace legacy VBA/Access applications with a modern, web-based platform featuring deep configurability, multi-tenancy, and integrated business intelligence.

### **Technical Architecture Summary**
- **Frontend:** Next.js 14 (App Router) hosted on Replit, utilizing React Server Components and Server Actions.
- **Backend & Database:** Supabase (PostgreSQL) for all data persistence, authentication, and real-time features.
- **File Storage:** Dual-mode architecture (Supabase Storage for cloud, Local Gateway for on-premise).
- **Integrations:** BambooHR (HRIS), Stripe (billing), and blockchain (NFT minting).
- **Core Innovation:** A 13-dimensional Work Breakdown Structure (WBS) engine with templating and automatic file replication.

### **Guiding Principles**
1.  **Tenant Isolation:** Every data row is scoped by `tenant_id`. Row Level Security (RLS) is mandatory for all tables.
2.  **Configuration over Code:** Branding, WBS dimensions, and module availability are driven by database configuration.
3.  **Event-Driven Synchronization:** Changes in one module (e.g., BambooHR sync) propagate correctly to others (e.g., WBS assignments).
4.  **Operational Simplicity:** Client onboarding is a semi-automated, template-driven process.

```sql
-- 1. WBS_NODES: The Heart of the System
---------------------------------------------
CREATE TABLE wbs_nodes (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    project_id UUID NOT NULL, -- Foreign key added after projects table creation
    parent_id UUID REFERENCES wbs_nodes(id) ON DELETE CASCADE,

    code_display TEXT GENERATED ALWAYS AS (REPLACE(code_path::text, '_', '.')) STORED,

    -- CONTENT
    title TEXT NOT NULL,
    description TEXT,
    status TEXT DEFAULT 'not_started' CHECK (status IN ('not_started', 'in_progress', 'on_hold', 'completed', 'cancelled')),

    -- THE 13-DIMENSIONAL MATRIX
    -- Dimensions are stored as key-value pairs. The *definition* of allowed keys is in tenants.config->'wbsDimensions'.
    dimensions JSONB NOT NULL DEFAULT '{}',

    -- ESTIMATES & ACTUALS
    estimated_hours DECIMAL(10,2),
    estimated_cost DECIMAL(15,2),
    actual_hours DECIMAL(10,2),
    actual_cost DECIMAL(15,2),

    -- RELATIONSHIPS
    assigned_to UUID REFERENCES tenant_users(id),
    predecessor_id UUID REFERENCES wbs_nodes(id),

    -- METADATA
    order_index INTEGER,
    is_template_node BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    -- CONSTRAINTS & INDEXES
    CONSTRAINT wbs_nodes_code_path_unique UNIQUE (tenant_id, project_id, code_path)
);

-- Enable the powerful LTREE extension
CREATE EXTENSION IF NOT EXISTS LTREE;

-- CRITICAL INDEXES for Performance
CREATE INDEX idx_wbs_nodes_tenant_project ON wbs_nodes(tenant_id, project_id);
CREATE INDEX idx_wbs_nodes_parent ON wbs_nodes(parent_id);
CREATE INDEX idx_wbs_nodes_code_path_gist ON wbs_nodes USING GIST (code_path);
CREATE INDEX idx_wbs_nodes_assigned_to ON wbs_nodes(assigned_to);
CREATE INDEX idx_wbs_nodes_dimensions ON wbs_nodes USING GIN (dimensions);

---------------------------------------------
-- 2. WBS_NODE_ATTRIBUTES: For Extensible Custom Fields
---------------------------------------------
CREATE TABLE wbs_node_attributes (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    node_id UUID NOT NULL REFERENCES wbs_nodes(id) ON DELETE CASCADE,
    attribute_key TEXT NOT NULL, -- e.g., 'concrete_psi', 'inspection_date'
    attribute_value JSONB NOT NULL, -- Can be string, number, boolean, array
    attribute_type TEXT NOT NULL CHECK (attribute_type IN ('text', 'number', 'date', 'boolean', 'file')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(node_id, attribute_key)
);

---------------------------------------------
-- 3. AUTOMATIC CODE_PATH MANAGEMENT TRIGGER
---------------------------------------------
-- This complex function ensures code_path is always correct and updates all children when a parent changes.
CREATE OR REPLACE FUNCTION update_wbs_code_path()
RETURNS TRIGGER AS $$
DECLARE
    old_path LTREE;
    new_path LTREE;
BEGIN
    -- Generate new path for INSERT or parent change
    IF TG_OP = 'INSERT' OR (TG_OP = 'UPDATE' AND NEW.parent_id IS DISTINCT FROM OLD.parent_id) THEN
        IF NEW.parent_id IS NULL THEN
            -- Root node: use its own ID
            NEW.code_path = NEW.id::text::ltree;
        ELSE
            -- Child node: parent path + its own ID
            SELECT code_path || NEW.id::text INTO new_path FROM wbs_nodes WHERE id = NEW.parent_id AND tenant_id = NEW.tenant_id;
            IF new_path IS NULL THEN
                RAISE EXCEPTION 'Invalid parent_id or tenant mismatch';
            END IF;
            NEW.code_path = new_path;
        END IF;
    END IF;

    -- If path changed, recursively update all descendants
    IF TG_OP = 'UPDATE' AND NEW.code_path IS DISTINCT FROM OLD.code_path THEN
        UPDATE wbs_nodes
        SET code_path = NEW.code_path || subpath(code_path, nlevel(OLD.code_path) - 1)
        WHERE tenant_id = NEW.tenant_id
          AND code_path <@ OLD.code_path
          AND id != NEW.id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER wbs_code_path_trigger
    BEFORE INSERT OR UPDATE ON wbs_nodes
    FOR EACH ROW EXECUTE FUNCTION update_wbs_code_path();
```

```javascript
// /lib/wbs-engine.js
import { createClient } from './supabase-server';

export class WBSEngine {
    constructor(tenantId) {
        this.tenantId = tenantId;
        this.supabase = createClient();
    }

    // 1. FETCH FULL TREE WITH DIMENSIONS
    async getProjectTree(projectId, filters = {}) {
        // Build query using LTREE operators for efficiency
        let query = this.supabase
            .from('wbs_nodes')
            .select(`
                *,
                assigned_to:tenant_users!wbs_nodes_assigned_to_fkey (
                    id, email, profile
                ),
                attributes:wbs_node_attributes(*)
            `)
            .eq('tenant_id', this.tenantId)
            .eq('project_id', projectId)
            .order('code_path', { ascending: true });

        // Apply dimension filters (e.g., dimensions->>'trade' = 'Electrical')
        Object.entries(filters).forEach(([key, value]) => {
            if (value) query = query.eq(`dimensions->>${key}`, value);
        });

        const { data, error } = await query;
        if (error) throw new Error(`Failed to fetch WBS tree: ${error.message}`);

        // Convert flat array to nested tree structure
        return this._buildTree(data);
    }

    // 2. MOVE NODE (CHANGE PARENT) WITH FULL RECALCULATION
    async moveNode(nodeId, newParentId) {
        // Use a stored procedure for atomic operation
        const { data, error } = await this.supabase.rpc('move_wbs_node', {
            p_node_id: nodeId,
            p_new_parent_id: newParentId,
            p_tenant_id: this.tenantId
        });

        if (error) throw new Error(`Failed to move node: ${error.message}`);
        return data;
    }

    // 3. APPLY DIMENSION FILTERS & AGGREGATION
    async getDimensionReport(projectId, dimensionKeys) {
        // For each key, get unique values and aggregate metrics
        const report = {};
        for (const key of dimensionKeys) {
            const { data } = await this.supabase
                .from('wbs_nodes')
                .select(`dimensions->>${key}, estimated_cost, actual_cost, status`)
                .eq('tenant_id', this.tenantId)
                .eq('project_id', projectId)
                .not(`dimensions->>${key}`, 'is', null);

            // Aggregate by dimension value
            report[key] = this._aggregateDimensionData(data, key);
        }
        return report;
    }

    // 4. PRIVATE HELPER: BUILD NESTED TREE FROM FLAT ARRAY
    _buildTree(nodes) {
        const map = {};
        const roots = [];

        // First pass: create map of id -> node with children array
        nodes.forEach(node => {
            map[node.id] = { ...node, children: [] };
        });

        // Second pass: assign children to parents
        nodes.forEach(node => {
            if (node.parent_id && map[node.parent_id]) {
                map[node.parent_id].children.push(map[node.id]);
            } else {
                roots.push(map[node.id]);
            }
        });

        // Calculate depth and path for each node
        const calculateDepth = (node, depth = 0) => {
            node.depth = depth;
            node.children.forEach(child => calculateDepth(child, depth + 1));
        };
        roots.forEach(root => calculateDepth(root));

        return roots;
    }

    // 5. PRIVATE HELPER: AGGREGATE DIMENSION DATA
    _aggregateDimensionData(data, key) {
        const aggregation = {};
        data.forEach(row => {
            const value = row[`dimensions->>${key}`];
            if (!value) return;

            if (!aggregation[value]) {
                aggregation[value] = {
                    count: 0,
                    estimated_cost: 0,
                    actual_cost: 0,
                    by_status: {}
                };
            }

            aggregation[value].count++;
            aggregation[value].estimated_cost += parseFloat(row.estimated_cost || 0);
            aggregation[value].actual_cost += parseFloat(row.actual_cost || 0);

            const status = row.status || 'unknown';
            aggregation[value].by_status[status] = (aggregation[value].by_status[status] || 0) + 1;
        });

        return aggregation;
    }
}
```

### **2.3 Frontend WBS Builder Component**
**Objective:** Create a drag-and-drop, real-time WBS interface with dimension editing.

**Instructions:** "Create the file `/components/wbs/WbsBuilder.jsx` with the following implementation."

```jsx
// /components/wbs/WbsBuilder.jsx
'use client';

import { useState, useEffect, useCallback } from 'react';
import { DndContext, closestCenter, KeyboardSensor, PointerSensor, useSensor, useSensors, DragEndEvent } from '@dnd-kit/core';
import { arrayMove, SortableContext, sortableKeyboardCoordinates, verticalListSortingStrategy } from '@dnd-kit/sortable';
import { WbsNode } from './WbsNode';
import { DimensionPanel } from './DimensionPanel';

export function WbsBuilder({ projectId, tenantId }) {
    const [tree, setTree] = useState([]);
    const [selectedNode, setSelectedNode] = useState(null);
    const [dimensionSchema, setDimensionSchema] = useState([]);

    // Initialize sensors for drag-and-drop
    const sensors = useSensors(
        useSensor(PointerSensor, { activationConstraint: { distance: 5 } }),
        useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
    );

    // Fetch tree and dimension schema on load
    useEffect(() => {
        fetchTree();
        fetchDimensionSchema();
    }, [projectId]);

    const fetchTree = async () => {
        const res = await fetch(`/api/wbs/tree?projectId=${projectId}`);
        const data = await res.json();
        setTree(data);
    };

    const fetchDimensionSchema = async () => {
        // Get from tenant config
        const res = await fetch(`/api/tenant/config`);
        const config = await res.json();
        setDimensionSchema(config.wbsDimensions || []);
    };

    // Handle drag end - update parent relationship
    const handleDragEnd = async (event) => {
        const { active, over } = event;
        if (!over || active.id === over.id) return;

        try {
            await fetch(`/api/wbs/nodes/${active.id}/move`, {
                method: 'PATCH',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ newParentId: over.id })
            });
            fetchTree(); // Refresh
        } catch (error) {
            console.error('Failed to move node:', error);
        }
    };

    // Render tree recursively
    const renderNode = (node) => (
        <WbsNode
            key={node.id}
            node={node}
            isSelected={selectedNode?.id === node.id}
            onSelect={setSelectedNode}
            dimensionSchema={dimensionSchema}
        >
            {node.children.map(renderNode)}
        </WbsNode>
    );

    return (
        <div className="flex h-screen">
            {/* Left Panel: Tree View */}
            <div className="w-3/4 border-r overflow-auto">
                <DndContext
                    sensors={sensors}
                    collisionDetection={closestCenter}
                    onDragEnd={handleDragEnd}
                >
                    <SortableContext items={tree.map(n => n.id)} strategy={verticalListSortingStrategy}>
                        <div className="p-4">
                            {tree.map(renderNode)}
                        </div>
                    </SortableContext>
                </DndContext>
            </div>

            {/* Right Panel: Dimension Editor */}
            <div className="w-1/4 bg-gray-50 p-4">
                <DimensionPanel
                    node={selectedNode}
                    dimensionSchema={dimensionSchema}
                    onSave={async (updates) => {
                        await fetch(`/api/wbs/nodes/${selectedNode.id}`, {
                            method: 'PATCH',
                            body: JSON.stringify(updates)
                        });
                        fetchTree();
                    }}
                />
            </div>
        </div>
    );
}
```

```javascript
// /lib/template-service.js
import { createClient } from './supabase-server';
import { StorageRouter } from './storage-router';

export class TemplateService {
    constructor(tenantId) {
        this.tenantId = tenantId;
        this.supabase = createClient();
        this.storage = new StorageRouter(tenantId);
    }

    // MAIN INSTANTIATION METHOD
    async instantiateTemplate(templateId, projectData, userId) {
        const transaction = await this.supabase.rpc('begin_instantiation', {
            p_template_id: templateId,
            p_tenant_id: this.tenantId,
            p_user_id: userId
        });

        try {
            // 1. CREATE PROJECT SHELL
            const project = await this.createProjectFromTemplate(templateId, projectData);

            // 2. GENERATE WBS STRUCTURE (Recursive)
            const wbsNodes = await this.generateWbsFromTemplate(templateId, project.id);

            // 3. COPY TEMPLATE FILES (Dual-mode aware)
            const copiedFiles = await this.replicateTemplateFiles(templateId, project.id);

            // 4. SET DEFAULT ATTRIBUTES & RELATIONSHIPS
            await this.applyTemplateDefaults(templateId, project.id, wbsNodes);

            // 5. LOG SUCCESSFUL INSTANTIATION
            await this.logInstantiation({
                templateId,
                projectId: project.id,
                userId,
                snapshot: {
                    wbs_nodes_created: wbsNodes.length,
                    documents_copied: copiedFiles.length,
                    project_data: projectData
                }
            });

            return {
                success: true,
                project,
                wbsNodes,
                copiedFiles
            };

        } catch (error) {
            await this.supabase.rpc('rollback_instantiation');
            throw new Error(`Template instantiation failed: ${error.message}`);
        }
    }

    // FILE REPLICATION ENGINE
    async replicateTemplateFiles(templateId, targetProjectId) {
        // Get file associations from template
        const { data: template } = await this.supabase
            .from('project_templates')
            .select('file_associations')
            .eq('id', templateId)
            .single();

        const copiedFiles = [];
        const errors = [];

        for (const association of template.file_associations || []) {
            try {
                // Determine source path (could be in template bucket or URL)
                const sourcePath = association.source_path;

                // Determine target path in new project
                const targetPath = `projects/${targetProjectId}/${association.target_in_project}`;

                // Use StorageRouter to copy (handles cloud/local mode)
                const result = await this.storage.copyFile({
                    sourcePath,
                    targetPath,
                    metadata: {
                        original_template: templateId,
                        copied_at: new Date().toISOString()
                    }
                });

                copiedFiles.push({
                    original: sourcePath,
                    copy: result.path,
                    association
                });

            } catch (error) {
                errors.push({
                    association,
                    error: error.message
                });
                if (association.is_required) {
                    throw new Error(`Required file copy failed: ${association.source_path}`);
                }
            }
        }

        return { copiedFiles, errors };
    }

    // WBS GENERATION FROM TEMPLATE SCHEMA
    async generateWbsFromTemplate(templateId, projectId) {
        const { data: template } = await this.supabase
            .from('project_templates')
            .select('template_schema')
            .eq('id', templateId)
            .single();

        const rootNodes = template.template_schema?.wbsRootNodes || [];
        const createdNodes = [];

        // Recursive function to create node hierarchy
        const createNode = async (nodeSchema, parentId = null, codePath = '') => {
            const nodeData = {
                tenant_id: this.tenantId,
                project_id: projectId,
                parent_id: parentId,
                title: nodeSchema.title,
                description: nodeSchema.description,
                dimensions: nodeSchema.dimensions || {},
                estimated_hours: nodeSchema.estimated_hours,
                is_template_node: true,
                order_index: nodeSchema.order
            };

            const { data: node, error } = await this.supabase
                .from('wbs_nodes')
                .insert(nodeData)
                .select()
                .single();

            if (error) throw error;

            createdNodes.push(node);

            // Create child nodes recursively
            for (const childSchema of nodeSchema.children || []) {
                await createNode(childSchema, node.id);
            }

            return node;
        };

        // Create all root nodes
        for (const rootSchema of rootNodes) {
            await createNode(rootSchema);
        }

        return createdNodes;
    }

    // LOGGING FUNCTION
    async logInstantiation(logData) {
        await this.supabase
            .from('template_instantiations')
            .insert({
                tenant_id: this.tenantId,
                ...logData
            });
    }
```

### **5.2 Storage Router with Intelligent Copying**
**Objective:** Create `/lib/storage-router.js` that handles dual-mode storage and automatic file replication.

**Instructions:** "Create this comprehensive storage service."

```javascript
// /lib/storage-router.js
import { createClient } from './supabase-server';
import { createReadStream, createWriteStream } from 'fs';
import { pipeline } from 'stream/promises';
import crypto from 'crypto';

export class StorageRouter {
    constructor(tenantId) {
        this.tenantId = tenantId;
        this.supabase = createClient();
    }

    // MAIN FILE COPY METHOD
    async copyFile(options) {
        const {
            sourcePath,
            targetPath,
            metadata = {},
            encryptionKey = null
        } = options;

        // 1. Determine source storage mode
        const sourceInfo = await this.analyzePath(sourcePath);
        
        // 2. Determine target storage mode (from tenant config)
        const targetMode = await this.getTenantStorageMode();
        
        // 3. Perform the copy based on mode combinations
        let result;
        if (sourceInfo.mode === 'cloud' && targetMode === 'cloud') {
            result = await this.copyCloudToCloud(sourceInfo, targetPath);
        } else if (sourceInfo.mode === 'local' && targetMode === 'cloud') {
            result = await this.copyLocalToCloud(sourceInfo, targetPath);
        } else if (sourceInfo.mode === 'cloud' && targetMode === 'local') {
            result = await this.copyCloudToLocal(sourceInfo, targetPath);
        } else {
            result = await this.copyLocalToLocal(sourceInfo, targetPath);
        }

        // 4. Encrypt if requested
        if (encryptionKey) {
            result = await this.encryptFile(result.finalPath, encryptionKey);
        }

        // 5. Create document record
        const docRecord = await this.createDocumentRecord({
            path: result.finalPath,
            mode: targetMode,
            metadata: {
                ...metadata,
                copied_from: sourcePath,
                copied_at: new Date().toISOString()
            }
        });

        return {
            ...result,
            documentId: docRecord.id
        };
    }

    // CLOUD-TO-CLOUD COPY (Supabase/S3)
    async copyCloudToCloud(sourceInfo, targetPath) {
        const { supabase } = this;

        // Download to buffer
        const { data: fileData, error: downloadError } = await supabase.storage
            .from(sourceInfo.bucket)
            .download(sourceInfo.path);

        if (downloadError) throw downloadError;

        // Upload to target
        const targetBucket = this.getTargetBucket();
        const { error: uploadError } = await supabase.storage
            .from(targetBucket)
            .upload(targetPath, fileData, {
                contentType: sourceInfo.mimeType,
                cacheControl: '3600'
            });

        if (uploadError) throw uploadError;

        return {
            finalPath: targetPath,
            mode: 'cloud',
            bucket: targetBucket,
            url: this.getPublicUrl(targetBucket, targetPath)
        };
    }

    // LOCAL-TO-CLOUD COPY (Gateway to Cloud)
    async copyLocalToCloud(sourceInfo, targetPath) {
        // Call local gateway API
        const gatewayEndpoint = await this.getGatewayEndpoint();
        
        const response = await fetch(`${gatewayEndpoint}/api/file/read`, {
            method: 'POST',
            headers: {
                'X-API-Key': await this.getGatewayApiKey(),
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                path: sourceInfo.fullPath,
                operation: 'read_for_upload'
            })
        });

        if (!response.ok) throw new Error('Gateway read failed');

        const { fileData, fileName, mimeType } = await response.json();

        // Upload to cloud
        const targetBucket = this.getTargetBucket();
        const { error } = await this.supabase.storage
            .from(targetBucket)
            .upload(targetPath, fileData, {
                contentType: mimeType
            });

        if (error) throw error;

        return {
            finalPath: targetPath,
            mode: 'cloud',
            bucket: targetBucket
        };
    }

    // AUTOMATIC TEMPLATE FILE REPLICATION
    async replicateTemplateFiles(templateId, targetProjectId) {
        // Get all template assets
        const { data: templateAssets } = await this.supabase
            .from('documents')
            .select('*')
            .eq('tenant_id', this.tenantId)
            .eq('template_id', templateId)
            .eq('is_template_asset', true);

        const results = [];

        for (const asset of templateAssets) {
            const targetPath = `projects/${targetProjectId}/templates/${asset.file_name}`;
            
            try {
                const result = await this.copyFile({
                    sourcePath: asset.storage_identifier,
                    targetPath,
                    metadata: {
                        original_asset_id: asset.id,
                        template_id: templateId,
                        generation: 1
                    }
                });

                results.push({
                    success: true,
                    original: asset,
                    copy: result
                });

            } catch (error) {
                results.push({
                    success: false,
                    original: asset,
                    error: error.message
                });
            }
        }

        return results;
    }

    // FILE ENCRYPTION FOR SENSITIVE DOCUMENTS
    async encryptFile(filePath, encryptionKey) {
        const algorithm = 'aes-256-gcm';
        const iv = crypto.randomBytes(16);
        
        const cipher = crypto.createCipheriv(algorithm, encryptionKey, iv);
        
        // Implementation depends on storage mode
        // This would involve reading, encrypting, and re-writing the file
        
        return {
            encrypted: true,
            iv: iv.toString('hex'),
            algorithm
        };
    }

    // PATH ANALYSIS HELPER
    async analyzePath(path) {
        // Determine if path is cloud URL, local path, or gateway reference
        if (path.startsWith('http')) {
            return {
                mode: 'cloud',
                url: path,
                bucket: this.extractBucketFromUrl(path),
                path: this.extractPathFromUrl(path)
            };
        } else if (path.startsWith('/') || path.includes('\\')) {
            return {
                mode: 'local',
                fullPath: path
            };
        } else {
            // Assume it's already a storage identifier
            const { data: doc } = await this.supabase
                .from('documents')
                .select('storage_mode, storage_identifier')
                .eq('storage_identifier', path)
                .single();

            return {
                mode: doc?.storage_mode || 'cloud',
                path: doc?.storage_identifier || path
            };
        }

            .from('tenants')
            .select('storage_mode')
            .eq('id', this.tenantId)
            .single();
        
        return data?.storage_mode || 'cloud';
    }

    async getGatewayEndpoint() {
        const { data } = await this.supabase
            .from('tenants')
            .select('gateway_endpoint')
            .eq('id', this.tenantId)
            .single();
        
        return data?.gateway_endpoint;
```

```javascript
// /lib/billing-service.js
import Stripe from 'stripe';
import { createClient } from './supabase-server';

export class BillingService {
    constructor(tenantId) {
        this.tenantId = tenantId;
        this.supabase = createClient();
        this.stripe = new Stripe(process.env.STRIPE_SECRET_KEY);
    }

    // AUTOMATIC SEAT ALLOCATION ON USER CREATION
    async allocateSeatForUser(userId, allocationType = 'primary') {
        const transaction = await this.supabase.rpc('begin_transaction');

        try {
            // 1. Get tenant's active subscription
            const subscription = await this.getActiveSubscription();
            if (!subscription) {
                throw new Error('No active subscription for tenant');
            }

            // 2. Check seat availability
            if (subscription.occupied_seats >= subscription.purchased_seats) {
                // Attempt to automatically add a seat
                const updated = await this.addSeatToSubscription();
                if (!updated) {
                    throw new Error('No seats available and auto-add failed');
                }
            }

            // 3. Allocate the seat
            const { error } = await this.supabase
                .from('seat_allocations')
                .insert([{
                    tenant_id: this.tenantId,
                    subscription_id: subscription.id,
                    user_id: userId,
                    allocation_type: allocationType
                }]);

            if (error) throw error;

            // 4. Update occupied seat count
            await this.updateOccupiedSeatCount(subscription.id);

            await this.supabase.rpc('commit_transaction');
            return true;

        } catch (error) {
            await this.supabase.rpc('rollback_transaction');
            throw error;
        }
    }

    // AUTOMATIC SEAT MANAGEMENT (CRON JOB)
    async reconcileSeats() {
        const subscription = await this.getActiveSubscription();
        if (!subscription) return;

        // Count active seat allocations
        const { count: occupiedSeats } = await this.supabase
            .from('seat_allocations')
            .select('*', { count: 'exact', head: true })
            .eq('tenant_id', this.tenantId)
            .eq('is_active', true);

        // If different from subscription count, update
        if (occupiedSeats !== subscription.occupied_seats) {
            await this.supabase
                .from('subscriptions')
                .update({ occupied_seats: occupiedSeats })
                .eq('id', subscription.id);
        }

        // Check if we need to adjust purchased seats
        const neededSeats = Math.max(occupiedSeats, subscription.purchased_seats);
        if (neededSeats > subscription.purchased_seats) {
            await this.addSeatsToSubscription(neededSeats - subscription.purchased_seats);
        }
    }

    // ADD SEATS WITH PRORATION
    async addSeatsToSubscription(quantity = 1) {
        const subscription = await this.getActiveSubscription();
        
        // Update Stripe subscription
        const stripeSub = await this.stripe.subscriptions.update(
            subscription.stripe_subscription_id,
            {
                items: [{
                    id: subscription.stripe_item_id,
                    quantity: subscription.purchased_seats + quantity
                }],
                proration_behavior: 'create_prorations'
            }
        );

        // Update local record
        await this.supabase
            .from('subscriptions')
            .update({
                purchased_seats: stripeSub.items.data[0].quantity,
                updated_at: new Date().toISOString()
            })
            .eq('id', subscription.id);

        return stripeSub;
    }

    // WEBHOOK HANDLER FOR STRIPE EVENTS
    async handleStripeWebhook(event) {
        // Store raw event
        await this.supabase
            .from('billing_events')
            .insert([{
                tenant_id: await this.getTenantIdFromStripeEvent(event),
                event_type: event.type,
                stripe_event_id: event.id,
                event_data: event.data.object,
                processed: false
            }]);

        // Process based on event type
        switch (event.type) {
            case 'customer.subscription.updated':
                await this.handleSubscriptionUpdate(event.data.object);
                break;
            case 'invoice.payment_succeeded':
                await this.handlePaymentSuccess(event.data.object);
                break;
            case 'invoice.payment_failed':
                await this.handlePaymentFailure(event.data.object);
                break;
        }

        // Mark as processed
        await this.supabase
            .from('billing_events')
            .update({ processed: true })
            .eq('stripe_event_id', event.id);
    }

    // HANDLE SUBSCRIPTION QUANTITY CHANGES FROM STRIPE
    async handleSubscriptionUpdate(stripeSubscription) {
        const { data: subscription } = await this.supabase
            .from('subscriptions')
            .select('*')
            .eq('stripe_subscription_id', stripeSubscription.id)
            .single();

        if (subscription) {
            // Extract quantity from Stripe
            const purchasedSeats = stripeSubscription.items.data[0].quantity;
            
            await this.supabase
                .from('subscriptions')
                .update({
                    purchased_seats: purchasedSeats,
                    stripe_status: stripeSubscription.status,
                    current_period_end: new Date(stripeSubscription.current_period_end * 1000),
                    updated_at: new Date().toISOString()
                })
                .eq('id', subscription.id);

            // If seats were reduced, may need to deallocate some users
            if (purchasedSeats < subscription.purchased_seats) {
                await this.handleSeatReduction(subscription.id, purchasedSeats);
            }
        }
    }

    // DEALLOCATE SEATS WHEN SUBSCRIPTION DOWNSIZED
    async handleSeatReduction(subscriptionId, newSeatCount) {
        // Get excess active allocations (oldest first)
        const { data: excessAllocations } = await this.supabase
            .from('seat_allocations')
            .select('*')
            .eq('subscription_id', subscriptionId)
            .eq('is_active', true)
            .order('allocated_at', { ascending: true })
            .range(newSeatCount, 999); // Get all beyond new limit

        // Deallocate the excess
        for (const allocation of excessAllocations) {
            await this.supabase
                .from('seat_allocations')
                .update({ deallocated_at: new Date().toISOString() })
                .eq('id', allocation.id);

            // Optionally notify user or admin
            await this.notifySeatDeallocation(allocation.user_id);
        }
    }

    // UTILITY METHODS
    async getActiveSubscription() {
        const { data } = await this.supabase
            .from('subscriptions')
            .select('*')
            .eq('tenant_id', this.tenantId)
            .eq('stripe_status', 'active')
            .order('created_at', { ascending: false })
            .limit(1)
            .single();

        return data;
    }

    async getTenantIdFromStripeEvent(event) {
        // Extract from event metadata or customer ID
        const customerId = event.data.object.customer;
        
        const { data } = await this.supabase
            .from('subscriptions')
            .select('tenant_id')
            .eq('stripe_customer_id', customerId)
            .single();

        return data?.tenant_id;
    }

    async updateOccupiedSeatCount(subscriptionId) {
        const { count } = await this.supabase
            .from('seat_allocations')
            .select('*', { count: 'exact', head: true })
            .eq('subscription_id', subscriptionId)
            .eq('is_active', true);

        await this.supabase
            .from('subscriptions')
            .update({ occupied_seats: count })
            .eq('id', subscriptionId);
    }

    async notifySeatDeallocation(userId) {
        // Implementation for email/SMS notification
        console.log(`User ${userId} deallocated from seat`);
    }
}
```

### **8.2 Data Encryption at Rest & in Transit**

**Objective:** Implement comprehensive encryption.

**Instructions:** "Create `/lib/encryption.js`"

```javascript
// lib/encryption.js
import crypto from 'crypto';
import { createClient } from './supabase-server';

export class EncryptionService {
    constructor() {
        this.algorithm = 'aes-256-gcm';
        this.key = Buffer.from(process.env.ENCRYPTION_SECRET, 'hex');
    }

    // ENCRYPT SENSITIVE DATA BEFORE DATABASE INSERTION
    async encryptDatabaseField(fieldValue, context = {}) {
        const iv = crypto.randomBytes(16);
        const cipher = crypto.createCipheriv(this.algorithm, this.key, iv);
        
        let encrypted = cipher.update(fieldValue, 'utf8', 'hex');
        encrypted += cipher.final('hex');
        
        const authTag = cipher.getAuthTag();
        
        return {
            encrypted: true,
            version: '1.0',
            algorithm: this.algorithm,
            iv: iv.toString('hex'),
            authTag: authTag.toString('hex'),
            ciphertext: encrypted,
            context: context
        };
    }

    // DECRYPT DATABASE FIELD
    async decryptDatabaseField(encryptedData) {
        if (!encryptedData.encrypted) return encryptedData;
        
        const decipher = crypto.createDecipheriv(
            this.algorithm,
            this.key,
            Buffer.from(encryptedData.iv, 'hex')
        );
        
        decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));
        
        let decrypted = decipher.update(encryptedData.ciphertext, 'hex', 'utf8');
        decrypted += decipher.final('utf8');
        
        return decrypted;
    }

    // ENCRYPT FILE STREAM
    async encryptFileStream(inputStream, outputStream) {
        const iv = crypto.randomBytes(16);
        const cipher = crypto.createCipheriv(this.algorithm, this.key, iv);
        
        // Write IV to beginning of file
        outputStream.write(iv);
        
        // Pipe through cipher
        inputStream.pipe(cipher).pipe(outputStream);
        
        // Return auth tag for storage in metadata
        return cipher.getAuthTag();
    }

    // COLUMN-LEVEL ENCRYPTION WRAPPER FOR SUPABASE
    static createEncryptedColumn(columnName) {
        return {
            get: async (value) => {
                const service = new EncryptionService();
                return service.encryptDatabaseField(value, { column: columnName });
            },
            set: async (encryptedValue) => {
                if (!encryptedValue) return null;
                const service = new EncryptionService();
                return service.decryptDatabaseField(encryptedValue);
            }
        };
    }
}
```
