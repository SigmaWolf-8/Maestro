  /**
   * Validate the query before processing

    if (!query.tenantId) {
      throw new AiReportError(
        'Tenant ID is required',
        AiReportErrorCode.INVALID_QUERY,
        query.id,
        query.tenantId
      );
    }
    
    if (!query.prompt || query.prompt.trim().length === 0) {
      throw new AiReportError(
        'Query prompt cannot be empty',
        AiReportErrorCode.INVALID_QUERY,
        query.id,
        query.tenantId
      );
    }
    
    if (query.prompt.length > (this.config.maxQueryLength || 1000)) {
      throw new AiReportError(
        `Query too long (max ${this.config.maxQueryLength} characters)`,
        AiReportErrorCode.INVALID_QUERY,
        query.id,
        query.tenantId
      );
    }
    
    // Check for potentially harmful prompts (basic check)
    const harmfulPatterns = [/drop table/i, /delete from/i, /insert into/i, /update.*set/i];
    if (harmfulPatterns.some(pattern => pattern.test(query.prompt))) {
      throw new AiReportError(
        'Query contains potentially harmful patterns',
        AiReportErrorCode.SECURITY_VIOLATION,
        query.id,
        query.tenantId
      );
    }
  }

  requiresWitnessing: z.boolean().default(false),
  context: z.object({
    projectId: z.string().optional(),
    dateRange: z.object({
      start: z.string().datetime(),
      end: z.string().datetime()
    }).optional(),
    costCenters: z.array(z.string()).optional(),
    wbsCodes: z.array(z.string()).optional()
  }).optional(),
  options: z.object({
    includeRawData: z.boolean().default(false),
    maxVisualizations: z.number().min(1).max(10).default(3),
    language: z.string().default('en'),
    detailLevel: z.enum(['summary', 'detailed', 'executive']).default('detailed'),
    format: z.enum(['interactive', 'pdf', 'excel']).default('interactive')
  }).optional()
});

export function createAiReportRouter(aiReportService: AiReportService): Router {
  const router = Router();

  /**
   * Generate a new AI report
   * POST /api/ai-report/generate
   */
  router.post('/generate', async (req, res) => {
    try {
      // Extract tenant and user from auth middleware (should be added by your auth middleware)
      const tenantId = req.tenantId; // Set by your tenant middleware
      const userId = req.userId;     // Set by your auth middleware
      
      if (!tenantId || !userId) {
        return res.status(401).json({
          error: 'Authentication required',
          code: 'AUTH_REQUIRED'
        });
      }

      // Validate request body
      const validated = GenerateReportSchema.parse(req.body);
      
      // Create query object
      const query: Omit<AiReportQuery, 'id' | 'timestamp'> = {
        tenantId,
        userId,
        prompt: validated.prompt,
        securityMode: validated.securityMode as AiReportSecurityMode,
        requiresWitnessing: validated.requiresWitnessing,
        context: validated.context
      };

      // Generate report
      const report = await aiReportService.generateReport(query, validated.options);
      
      // Return success response
      return res.status(200).json({
        success: true,
        data: report,
        metadata: {
          queryId: report.queryId,
          generatedAt: report.generatedAt,
          generationTimeMs: report.generationTimeMs
        }
      });

    } catch (error) {
      // Handle validation errors
      if (error instanceof z.ZodError) {
        return res.status(400).json({
          error: 'Validation failed',
          details: error.errors,
          code: 'VALIDATION_ERROR'
        });
      }
      
      // Handle AI report errors
      if (error instanceof AiReportError) {
        return res.status(400).json({
          error: error.message,
          code: error.code,
          queryId: error.queryId,
          tenantId: error.tenantId
        });
      }
      
      // Handle unexpected errors
      console.error('AI Report generation error:', error);
      return res.status(500).json({
        error: 'Internal server error',
        code: 'INTERNAL_ERROR'
      });
    }
  });

  /**
   * Get service health status
   * GET /api/ai-report/health
   */
  router.get('/health', async (_req, res) => {
    try {
      const health = await aiReportService.getHealthStatus();
      return res.status(200).json(health);
    } catch (error) {
      return res.status(500).json({
        status: 'unhealthy',
        message: 'Service health check failed',
        timestamp: new Date()
      });
    }
  });

  /**
   * Get tenant usage statistics
   * GET /api/ai-report/usage
   */
  router.get('/usage', async (req, res) => {
    try {
      const tenantId = req.tenantId;
      if (!tenantId) {
        return res.status(401).json({ error: 'Tenant ID required' });
      }
      
      const usage = await aiReportService.getTenantUsage(tenantId);
      return res.status(200).json(usage);
    } catch (error) {
      return res.status(500).json({
        error: 'Failed to fetch usage statistics',
        code: 'USAGE_FETCH_FAILED'
      });
    }
  });

  return router;
}
