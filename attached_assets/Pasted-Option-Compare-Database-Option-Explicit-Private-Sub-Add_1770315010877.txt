Private Sub ChangeName_Click()

If NewNameInput <> "" Then
    Screen.MousePointer = 11
    que = MsgBox("Are you sure that you would like to change this vendor name?" & vbCrLf & "Doing so, will change PO's, Quotes, Invoices, Price Master...", vbYesNo)
    If que = vbYes Then
        CurrentDb.Execute "UPDATE SalviVendors SET SalviVendors.Company ='" & NewNameInput & "' WHERE (((SalviVendors.Company)='" & Me.Vendor & "'));"

        CurrentDb.Execute "UPDATE PM SET PM.Vendor ='" & NewNameInput & "' WHERE (((PM.Vendor)='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE PM SET PM.Vendor ='" & NewNameInput & "' WHERE (((PM.Vendor)='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE PMCompile SET PMCompile.Vendor ='" & NewNameInput & "' WHERE (((PMCompile.Vendor)='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE PMCompile SET PMCompile.SubVendor ='" & NewNameInput & "' WHERE (((PMCompile.SubVendor)='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE PONumbers SET PONumbers.[Supplier/Trade] ='" & NewNameInput & "' WHERE (((PONumbers.[Supplier/Trade])='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE PORelations SET PORelations.[Supplier/Trade] ='" & NewNameInput & "' WHERE (((PORelations.[Supplier/Trade])='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE Quotes SET Quotes.Customer ='" & NewNameInput & "' WHERE (((Quotes.Customer)='" & Me.Vendor & "'));"
        CurrentDb.Execute "UPDATE INVNumbers SET INVNumbers.[Supplier/Trade] ='" & NewNameInput & "' WHERE (((INVNumbers.[Supplier/Trade])='" & Me.Vendor & "'));"

            Me.Vendor.Value = NewNameInput

Private Sub Form_BeforeInsert(Cancel As Integer)
Dim Yoda As DAO.Database, rs As DAO.Recordset


    Set Yoda = CurrentDb
    Set rs = Yoda.OpenRecordset("Select * From SalviContacts Order by ContactID", dbOpenSnapshot)

    If Not rs.EOF Then
        rs.MoveLast
        Me.ContactID.Value = rs!ContactID + 1
    End If

If Me.Vendor = "Salvi Homes" Then
        Set rs = Yoda.OpenRecordset("Select * From Employees", dbOpenDynaset)
            If Not rs.EOF Then
                rs.AddNew
                rs!ContactID = Me.ContactID.Value
                rs.Update
            End If
End If

    rs.Close
    Yoda.Close
    Set Yoda = Nothing
    Set rs = Nothing

End Sub

Private Sub Vendor_KeyDown(KeyCode As Integer, Shift As Integer)
If PassWhat = "SuperAdmin" Or PassWhat = "Admin" Then

Dim ControlDown As Integer, CD As Integer, Warn As Integer, LastChance As Integer
    ControlDown = (Shift And acShiftMask) > 0
    CD = (Shift And acCtrlMask) > 1

If CD And KeyCode = vbKeyD Then
    Warn = MsgBox("Holding down Ctrl and the letter D at the same time will Delete this Vendor AND ALL CONTACTS WITHIN THIS COMPANY" & vbCrLf & "Do you wish to continue", vbYesNo, "Delete Vendor")
    If Warn = vbYes Then
        CurrentDb.Execute "Delete * From [SalviVendors] Where [Company]='" & Me.Vendor.Value & "'"
        AutoSendEmail AccountantE & ";" & PresidentE, , , "A Vendor has been deleted from 'The Maestro'", "Please delete this vendor from Peachtree- " & Me.Company & " (Vendor ID-)" & Me.Vendor_ID, , , , , , , True
        LastChance = MsgBox("Last chance are you sure you wish to also delete ALL CONTACTS WITHIN THIS COMPANY" & vbCrLf & "Do you wish to continue", vbYesNo, "Delete Vendor Contacts")
            If LastChance = vbYes Then
                CurrentDb.Execute "Delete * From [SalviContacts] Where [Company]='" & Me.Vendor.Value & "'"
            End If
        Me.Vendor.Value = ""
        Me.Requery
    End If
SendKeys "{ESC}"
End If

End If
End Sub

Public Sub SetRatings()
Dim Counter As Integer, Rater As Variant
Dim reliability As Variant, Quality As Variant, Speed As Variant, Pricing As Variant, Congeniality As Variant

reliability = Nz(Me.RateRE, 0)
If reliability <> "" Then reliability = 1 Else reliability = 0

Quality = Nz(Me.RateQU, 0)
If Quality <> "" Then Quality = 1 Else Quality = 0

Speed = Nz(Me.RateSP, 0)
If Speed <> "" Then Speed = 1 Else Speed = 0

Pricing = Nz(Me.RatePR, 0)
If Pricing <> "" Then Pricing = 1 Else Pricing = 0

Congeniality = Nz(Me.RateCO, 0)
If Congeniality <> "" Then Congeniality = 1 Else Congeniality = 0

Counter = reliability + Quality + Speed + Pricing + Congeniality

reliability = Nz(Me.RateRE, 0)
    If reliability = "" Then reliability = 0
Quality = Nz(Me.RateQU, 0)
    If Quality = "" Then Quality = 0
Speed = Nz(Me.RateSP, 0)
    If Speed = "" Then Speed = 0
Pricing = Nz(Me.RatePR, 0)
    If Pricing = "" Then Pricing = 0
Congeniality = Nz(Me.RateCO, 0)
    If Congeniality = "" Then Congeniality = 0

Rater = (reliability + Quality + Speed + Pricing + Congeniality)

Debug.Print "Reliability-" & reliability
Debug.Print "Quality-" & Quality
Debug.Print "Speed-" & Speed
Debug.Print "Pricing-" & Pricing
Debug.Print "Congeniality-" & Congeniality
Debug.Print "Total Rating-" & Rater

'MsgBox Counter


If Counter > 0 Then
    If Rater <> "" Then Rater = Round((Rater / Counter), 1) Else Rater = 0
End If

If Rater < 3 Then Me.Rating = Rater & " Rating (Below Threshold)" Else Me.Rating = Rater & " Rating"
If Rater = 0 Then Me.Rating = "No Rating"
End Sub
