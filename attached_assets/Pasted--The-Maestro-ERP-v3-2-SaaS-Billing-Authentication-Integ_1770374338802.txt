### 4.1 `server/services/ledger-witness-service.ts` [NEW] (~120 lines)

This service replaces all XRPL witnessing in the SaaS layer with Algorand (primary) and Hedera (fallback).

```typescript
/**
 * Ledger Witness Service
 * 
 * Provides immutable witnessing of critical business events (subscription creation,
 * payment processing, security mode changes) via Algorand (primary) and Hedera (fallback).
 * 
 * IMPORTANT: This does NOT replace the existing PlenumNET XRPL witnessing in
 * server/integrations/plenum-net-core-client.ts — that remains for PlenumNET-internal
 * operations. This service handles SaaS billing/subscription witnessing exclusively.
 * 
 * Implementation Notes:
 * - Use algosdk (npm: algosdk) for Algorand interactions
 * - Use @hashgraph/sdk for Hedera interactions
 * - Both SDKs support testnet for development
 * - Witness receipts are stored in subscription_invoices table columns
 * - Environment variables: ALGORAND_TOKEN, ALGORAND_SERVER, ALGORAND_PORT,
 *   HEDERA_ACCOUNT_ID, HEDERA_PRIVATE_KEY, HEDERA_NETWORK
 */

export interface WitnessPayload {
  eventType: "subscription_created" | "payment_processed" | "plan_changed" | "security_mode_changed" | "tenant_provisioned";
  tenantId: number;
  timestamp: Date;
  data: Record<string, unknown>;
}

export interface WitnessResult {
  provider: "algorand" | "hedera";
  success: boolean;
  transactionId: string;
  // Algorand
  confirmedRound?: number;
  // Hedera
  consensusTimestamp?: string;
  topicSequenceNumber?: number;
}

export class LedgerWitnessService {
  /**
   * Witness an event on Algorand. Falls back to Hedera on failure.
   * 
   * Algorand approach: Use application call to a deployed smart contract
   * that stores the SHA-256 hash of the witness payload as a note.
   * 
   * Hedera approach: Submit message to a Hedera Consensus Service (HCS) topic.
   */
  async witnessEvent(payload: WitnessPayload): Promise<WitnessResult>;
  
  /**
   * Verify a witness receipt against the ledger.
   */
  async verifyWitness(receipt: WitnessResult): Promise<boolean>;
  
  /**
```

```typescript
/**
 * Entitlement Middleware
 * 
 * Express middleware that gates API endpoints by subscription plan and security mode.
 * Attaches entitlement context to the request object for downstream handlers.
 * 
 * Usage in route files:
 *   import { featureGuard } from "../middleware/entitlement";
 *   router.get("/smart-inbox", featureGuard("smart_inbox", "phi"), handler);
 * 
 * CRITICAL: This middleware must NOT be applied to existing v3.1 routes.
 * Only new v3.2 routes and explicitly flagged endpoints should use it.
 * Existing endpoints continue to work with the current auth-only middleware.
 */

import type { Request, Response, NextFunction } from "express";
import type { SecurityMode } from "../../shared/types/subscriptions";
import { SECURITY_MODE_REQUIREMENTS, PLAN_REQUIREMENTS } from "../../shared/types/subscriptions";

// Extends Express Request
declare global {
  namespace Express {
    interface Request {
      entitlement?: {
        tenantId: number;
        planCode: string;
        planName: string;
        status: string;
        securityMode: SecurityMode;
        features: Record<string, boolean>;
      };
    }
  }
}

/**
 * Creates an Express middleware that checks:
 * 1. Tenant has an active subscription
 * 2. Subscription plan includes the required feature
 * 3. Security mode meets the required level (if specified)
 * 4. Usage is within plan limits
 * 
 * Returns 403 with upgrade suggestion on failure.
 * Returns 429 if usage limits exceeded.
 */
export function featureGuard(
  requiredFeature: string,
  requiredSecurityMode?: SecurityMode
): (req: Request, res: Response, next: NextFunction) => Promise<void>;

/**
 * Lightweight middleware that only checks for active subscription.
 * Use for billing/subscription management routes themselves.
 */
export function requireActiveSubscription(): (req: Request, res: Response, next: NextFunction) => Promise<void>;

 * Helper: Compare security modes. Returns true if `current` >= `required`.
 */
export function meetsSecurityLevel(current: SecurityMode, required: SecurityMode): boolean;
```
