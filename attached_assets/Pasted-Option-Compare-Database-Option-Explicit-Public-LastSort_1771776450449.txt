Private Sub Form_AfterInsert()

Public Sub PMCompileLineItemonForm()
'Me.Recalc
            If Me.ExpressionValue = "*" Then Me.Subtotal = Me.Quantity * Me.Price
            If Me.ExpressionValue = "/" Then Me.Subtotal = Me.Quantity / Me.Price
            If Me.ExpressionValue = "+" Then Me.Subtotal = Me.Quantity + Me.Price
            If Me.ExpressionValue = "-" Then Me.Subtotal = Me.Quantity - Me.Price
            If Nz(Me.ExpressionValue, "0") = "0" Then Me.Subtotal = Me.Price
            
            If Me.ExpressionValue2 = "*" Then Me.Subtotal2 = Me.Subtotal * Me.Quantity2
            If Me.ExpressionValue2 = "/" Then Me.Subtotal2 = Me.Subtotal / Me.Quantity2
            If Me.ExpressionValue2 = "+" Then Me.Subtotal2 = Me.Subtotal + Me.Quantity2
            If Me.ExpressionValue2 = "-" Then Me.Subtotal2 = Me.Subtotal - Me.Quantity2
            If Nz(Me.ExpressionValue2, "0") = "0" Then Me.Subtotal2 = Me.Subtotal
            
            If Me.ExpressionValue3 = "*" Then Me.LineTotal = Me.Subtotal2 * Me.Quantity3
            If Me.ExpressionValue3 = "/" Then Me.LineTotal = Me.Subtotal2 / Me.Quantity3
            If Me.ExpressionValue3 = "+" Then Me.LineTotal = Me.Subtotal2 + Me.Quantity3
            If Me.ExpressionValue3 = "-" Then Me.LineTotal = Me.Subtotal2 - Me.Quantity3
            If Nz(Me.ExpressionValue3, "0") = "0" Then Me.LineTotal = Me.Subtotal2
'Line item has been recalulated already
Me.PUP = False
SetPMPStoRecalculate Me.PSHeader, Me.Vendor

Me.Recalc
End Sub
Public Function GetPMPrice(PSSub As String, Vendor As String)
Dim Yoda As DAO.Database, rs As DAO.Recordset
Dim Newprice As Single
Set Yoda = CurrentDb

Set rs = Yoda.OpenRecordset("Select * from Pm where ps='" & PSSub & "' and vendor='" & Vendor & "'", dbOpenSnapshot)

If Not rs.EOF Then
    Newprice = Nz(rs!NewUpdate, Nz(rs!Price, 0))
    Me.Price = Newprice
End If

rs.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End Function
