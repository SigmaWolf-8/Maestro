Private Sub Address_AfterUpdate()
Dim Yoda As DAO.Database, rs As DAO.Recordset
Set Yoda = CurrentDb
Set rs = Yoda.OpenRecordset("Select * From Customers Where Job=" & Me.JobNum, dbOpenDynaset, adLockOptimistic)


If Not rs.EOF Then
    rs.Edit
    rs![Address] = Me.[Address]
    rs.Update
End If
Me.Recalc

rs.Close
Yoda.Close
Set rs = Nothing
Set Yoda = Nothing
End Sub

Private Sub Form_Open(Cancel As Integer)
If Forms.Password.PasswordBox = Pass And AllowEnter(Me.Name) = True Then Me.Visible = True Else Me.Visible = False
Me.AllowAdditions = AllowAdd(Me.Name)
Me.AllowEdits = AllowEdit(Me.Name)
Me.AllowDeletions = AllowDelete(Me.Name)

Me.BorderBox.BackColor = YodaBorderColour
Me.FormHeader.BackColor = YodaHeadColour
Me.FormFooter.BackColor = YodaFootColour


If PassWhat = "SuperAdmin" Or PassWhat = "Admin" Or PassWhat = "OManager" Then
    Me.EditFieldValues.Visible = True
    Me.AllowDeletions = True
Else
    Me.EditFieldValues.Visible = False
    Me.AllowDeletions = False
End If

Me.JobNum.SetFocus
End Sub

Public Sub JobNum_AfterUpdate()
Me.RecordSource = "Select Customers.* FROM Customers WHERE (((Customers.Job)= " & Me.JobNum & "));"
Dim Yoda As DAO.Database, rs As DAO.Recordset
Set Yoda = CurrentDb
Set rs = Yoda.OpenRecordset("SELECT Customers.Job, Customers.*, Quotes.* FROM Customers INNER JOIN Quotes ON Customers.Job = Quotes.JobNum WHERE (((Customers.Job)= " & Me.JobNum & "));", dbOpenSnapshot)

If Not rs.EOF Then
    Me.Address = Nz(rs!Address, "")
    Me.City = Nz(rs!City, "")
    Me.State_Province = Nz(rs![State/Province], "")
    Me.ZIP_Postal_Code = Nz(rs![ZIP/Postal Code], "")
    Me.Country = Nz(rs![Country/Region], "")
    Me.FirstNames = Nz(rs![First Name], "")
    Me.LastNames = Nz(rs![Last Name], "")
    Me.DisplayNames = Nz(rs![Customer], "")
    Me.WebPage = Nz(rs![Web Page], "")
    Me.HomePhone = Nz(rs![Home Phone], "")
    Me.QNum = Nz(rs![QNum], "")
    Me.DateofQuote = Nz(rs![DateofQuote], "")
    Me.Division = Nz(rs![Division], "")
    Me.Model = Nz(rs![Model], "")
    Me.ProjectAddress = Nz(rs![ProjectAddress], "")
    Me.Lot = Nz(rs![Lot], "")
    Me.Block = Nz(rs![Block], "")
    Me.Plan = Nz(rs![Plan], "")
    Me.Main = Nz(rs![Main], "")
    Me.Upper = Nz(rs![Upper], "")
    Me.Low = Nz(rs![Low], "")
    Me.Gar = Nz(rs![Gar], "")
    Me.DP = Nz(rs![DP], "")
    Me.BP = Nz(rs![BP], "")
    Me.DGBP = Nz(rs![DGBP], "")
Else
    Me.Address = ""
    Me.City = ""
    Me.State_Province = ""
    Me.ZIP_Postal_Code = ""
    Me.Country = ""
    Me.FirstNames = ""
    Me.LastNames = ""
