Public Function ARTerms(CompanyName As String) As Variant
Dim Yoda As DAO.Database, rs As DAO.Recordset
    Set Yoda = CurrentDb
    Set rs = Yoda.OpenRecordset("Select * from SalviVendors where Company='" & CompanyName & "'", dbOpenSnapshot)
    If Not rs.EOF Then
        ARTerms = Nz(rs![AR Terms], 7)
        If InStr(ARTerms, "net") Then
            ARTerms = Mid(rs![AR Terms], 5)
        End If

        If InStr(ARTerms, "on reciept") Then
            ARTerms = 0
        End If
    Else
        ARTerms = 7
    End If

rs.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End Function

Public Sub CreatePayrollGLAllocationBatch()
Dim NumDis As Integer, dateANDTime As String, Filename As String, TodayDate As String, Monthdate As String, Daydate As String
Dim Yoda As DAO.Database, rs As DAO.Recordset, fld As Field, ReferenceDate As String
Set Yoda = CurrentDb


Screen.MousePointer = 11


'Store Reference Invoice Number
Monthdate = Month(Now)
If Len(Monthdate) = 1 Then Monthdate = "0" & Monthdate
Daydate = Day(Now)
If Len(Daydate) = 1 Then Daydate = "0" & Daydate
ReferenceDate = Mid(Year(Now), 3, 2) & Monthdate & Daydate & "_" & Hour(Time) & ":" & Minute(Time) & ":" & Second(Time)


 
CurrentDb.Execute "SELECT #" & Date & "# As [Transaction Date],'" & ReferenceDate & "' as Reference, 1 as [Number of Distributions],Payroll.Code as [G/L Account],'Payroll Allocation #" & Date & "# -' & [Payroll]![Completed for ID]" & " AS Description, Sum(Payroll.GLAMT) AS Amount, Payroll![Completed for ID] As [Job ID] INTO PeachTreePayrollAllocationPostTemp FROM PeachTreePayrollPost LEFT JOIN Payroll ON PeachTreePayrollPost.[Ticket Number] = Payroll.[Ticket Number] GROUP BY Payroll.Code,[Payroll]![Completed for ID];"


'Need to revise 71010 to Hourly payroll account
CurrentDb.Execute "Insert into PeachTreePayrollAllocationPostTemp([Transaction Date],[Reference],Amount,[G/L Account],Description) SELECT #" & Date & "# as date1,'" & ReferenceDate & "' As Refdate, Sum(Payroll.GLAMT) AS Amount,'71010' as Expr1,'Total sum being allocated from time tickets' as Expr2 FROM PeachTreePayrollPost LEFT JOIN Payroll ON PeachTreePayrollPost.[Ticket Number] = Payroll.[Ticket Number];"
'Truncate phase and job codes if job number is missing
CurrentDb.Execute "UPDATE PeachTreePayrollAllocationPostTemp SET PeachTreePayrollAllocationPostTemp.[Job ID] = null WHERE (((PeachTreePayrollAllocationPostTemp.[Job ID]) Like (',*')));"
'Truncate Extraneous job codes used for database tracking
CurrentDb.Execute "Update PeachTreePayrollAllocationExported SET PeachTreePayrollAllocationExported.[Job ID] = Mid([JOb ID],1,Len([Job Id])-5) WHERE (((PeachTreePayrollAllocationExported.[Job ID]) Like '*.*'));"


'Count records
Set rs = Yoda.OpenRecordset("Select * From PeachTreePayrollAllocationPostTemp")
If Not rs.EOF Then
    rs.MoveLast
    NumDis = rs.RecordCount
End If
rs.Close
Set rs = Nothing


'Update table
CurrentDb.Execute "Update PeachTreePayrollAllocationPostTemp Set [Number of Distributions]=" & NumDis & ""


'Create Spreadsheet
TodayDate = Month(Now) & "_" & Day(Now) & "_" & Year(Now)
dateANDTime = " (" & Hour(Time) & "-" & Minute(Time) & "-" & Second(Time) & ")"
Filename = "Payroll_To_GLAccounts_" & TodayDate & dateANDTime
DoCmd.TransferText acExportDelim, , "PeachTreePayrollAllocationPostTemp", ExcelPayrollLocation & Filename & ".csv", True 'Text Format
         
        'Copy exported items into backupfile
    TodayDate = Now
    CurrentDb.Execute ("INSERT INTO PeachTreePayrollAllocationExported ( [Transaction Date], Reference, [Number of Distributions], [G/L Account], Description, Amount, [Job ID], ExportDate) SELECT PeachTreePayrollAllocationPostTemp.[Transaction Date], PeachTreePayrollAllocationPostTemp.Reference, PeachTreePayrollAllocationPostTemp.[Number of Distributions], PeachTreePayrollAllocationPostTemp.[G/L Account], PeachTreePayrollAllocationPostTemp.Description, PeachTreePayrollAllocationPostTemp.Amount, PeachTreePayrollAllocationPostTemp.[Job ID], #" & TodayDate & "# AS Exp1 FROM PeachTreePayrollAllocationPostTemp;")
    'Delete Temp File
    DoCmd.DeleteObject acTable, "PeachTreePayrollAllocationPostTemp"


Screen.MousePointer = 1


Yoda.Close
Set Yoda = Nothing


End Sub
Public Function INVPOTotal(PONum As Long, Optional USD As Boolean)
Dim Yoda As DAO.Database, rs As DAO.Recordset
Set Yoda = CurrentDb

Set rs = Yoda.OpenRecordset("Select Sum (amt) as Sumsub, PONum, POClosed From PONumbers Group by Ponum,POClosed Having PoNum=" & PONum & " And POClosed=False", dbOpenSnapshot)
If Not rs.EOF Then
    INVPOTotal = rs!SumSub
    If USD = True Then INVPOTotal = rs!SumSub * USDC
Else
    Set rs = Yoda.OpenRecordset("Select Sum (CADAmt) as Sumsub, PONum From INVap Where GLAccount<>'11150' Group by Ponum Having PoNum=" & PONum, dbOpenSnapshot)
    If Not rs.EOF Then INVPOTotal = Round(rs!SumSub, 2)
End If

Debug.Print INVPOTotal

rs.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End Function

Public Sub CarInventoryCost(Car As String)
Dim Yoda As DAO.Database, rs As DAO.Recordset, rsc As DAO.Recordset, TodayDate As String
Set Yoda = CurrentDb
Set rs = Yoda.OpenRecordset("SELECT BOLRail.CarNum, BOLRail.BOL, BOLRail.PO, BOLRail.OriginRL, BOLRail.Quantity, BOLRail.DestinationRL, Sum(POData.Quantity) AS SumOfQuantity, BOLRailPO.OriginRLPO, BOLRailPO.DestinationRLPO, BOLRailPO.CustomsPO, BOLRailPO.CNPO FROM POData INNER JOIN (BOLRail INNER JOIN BOLRailPO ON BOLRail.CarNum = BOLRailPO.CarNum) ON POData.PONum = BOLRail.PO GROUP BY BOLRail.CarNum, BOLRail.BOL, BOLRail.PO, BOLRail.OriginRL, BOLRail.Quantity, BOLRail.DestinationRL, BOLRailPO.Carnum, BOLRailPO.OriginRLPO, BOLRailPO.DestinationRLPO, BOLRailPO.CustomsPO, BOLRailPO.CNPO HAVING (((Sum(POData.Quantity))>0) AND ((BOLRailPO.Carnum)='" & Car & "'));", dbOpenSnapshot)

Set rsc = Yoda.OpenRecordset("Select * From MatInventoryCosts Where Carnum='" & Car & "'", dbOpenDynaset)
Dim OriginRL As Single, DestinationRL As Single, Customs As Single, Rail As Single, Mat As Single
Dim TotalCost As Single

MsgBox "Costs need to be generated and sent to accounting for landed inventory." & vbCrLf & "You may be prompted to enter some USD conversion estimates if all invoicing has not been recieved.", vbOKOnly, "Please be patient while The Core works...."

If Not rs.EOF Then
    OriginRL = INVPOTotal(rs!OriginRLPO, True)
    Rail = INVPOTotal(rs!CNPO, True)
    Customs = INVPOTotal(rs!CustomsPO)
    DestinationRL = INVPOTotal(Nz(rs!DestinationRLPO, 0))
    Mat = INVPOTotal(rs!PO, True)


    If rsc.EOF Then
        rsc.AddNew
        rsc!CarNum = Car
        rsc!OriginRL = OriginRL
        rsc!Rail = Rail
        rsc!Customs = Customs
        rsc!DestinationRL = DestinationRL
        rsc!Mats = (Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity")
        rsc!TotalCost = OriginRL + Rail + Customs + DestinationRL + (Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity")
        rsc!CostPerMat = (OriginRL + Rail + Customs + DestinationRL + ((Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity"))) / rs.Fields("Quantity")
        rsc.Update
    Else
        rsc.FindFirst "CarNum =" & "'" & Car & "'"
        If Not rsc.NoMatch Then
            rsc.Edit
            rsc!CarNum = Car
            rsc!OriginRL = OriginRL
            rsc!Rail = Rail
            rsc!Customs = Customs
            rsc!DestinationRL = DestinationRL
            rsc!Mats = (Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity")
            rsc!TotalCost = OriginRL + Rail + Customs + DestinationRL + (Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity")
            rsc!CostPerMat = (OriginRL + Rail + Customs + DestinationRL + ((Mat / rs.Fields("SumofQuantity")) * rs.Fields("Quantity"))) / rs.Fields("Quantity")
            rsc.Update
        End If
    End If

TodayDate = Month(Now) & "_" & Day(Now) & "_" & Year(Now)
DoCmd.OutputTo acOutputReport, "MatsToInventoryCosts", acFormatPDF, InventoryTranLocation & Round(rs.Fields("BOL"), 2) & "(" & TodayDate & ").pdf", False

End If

Public Function WeekofCurrPayperiod(Dater As Date) As Byte
Dim PBeg As Date, PEnd As Date
PBeg = CurrentPayPeriodBeg
PEnd = CurrentPayPeriodEnd

'Debug.Print DateAdd("d", -8, PEnd) & " End of Week 1"
'Debug.Print Dater & " Date Processed"

If Dater <= DateAdd("d", -8, PEnd) Then WeekofCurrPayperiod = 1 Else WeekofCurrPayperiod = 2
If Dater < PBeg Then WeekofCurrPayperiod = 0
If Dater >= PEnd Then WeekofCurrPayperiod = 0

End Function

'Set Rsinv = Yoda.OpenRecordset("SELECT INVNumbers.Job, Sum(Invdata.SubTotal) AS SumOfSubTotal, Invdata.NA, INVNumbers.Job FROM Invdata INNER JOIN INVNumbers ON Invdata.INVNum = INVNumbers.INVNum GROUP BY INVNumbers.Job, Invdata.NA HAVING (((INVNumbers.Job)=" & Me.Job & ") AND ((Invdata.[NA])=False));", dbOpenSnapshot)

If Not rs.EOF Then
Do Until rs.EOF
    rss.FindFirst "Job=" & Job
    If Not rss.NoMatch Then
        rss.Edit
        rss!TotalGrossCost = rs!TotalCost
        rss!Profit = rss!CustomerInvoiceAmt - rss!TotalGrossCost
        If rss!CustomerInvoiceAmt > 0 Then rss!GrossMargin = Round(rss!Profit / Nz(rss!CustomerInvoiceAmt, 0), 4)
        rss.Update
    Else
        rss.AddNew
        rss!Job = Job
        rss!TotalGrossCost = rs!TotalCost
        rss!Profit = rss!CustomerInvoiceAmt - rss!TotalGrossCost
        If rss!CustomerInvoiceAmt > 0 Then rss!GrossMargin = Round(rss!Profit / Nz(rss!CustomerInvoiceAmt, 0), 4)
        rss.Update
    End If

rs.MoveNext
Loop
End If


rs.Close
rss.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
Set rss = Nothing
End Sub
Public Function CurrentBurden(WhatBurden As String) As Single
Dim Yoda As DAO.Database, rs As DAO.Recordset
Set Yoda = CurrentDb


Set rs = Yoda.OpenRecordset("Select * from BurdenRates", dbOpenSnapshot)
     
    If WhatBurden = "CPP" Then CurrentBurden = rs!CPP
    If WhatBurden = "EI" Then CurrentBurden = rs!EI
    If WhatBurden = "WCB" Then CurrentBurden = rs!WCB
    If WhatBurden = "LiabilityInsurance" Then CurrentBurden = rs!LiabilityInsurance
    If WhatBurden = "VacPay" Then CurrentBurden = rs!VacPay
    If WhatBurden = "Safety" Then CurrentBurden = rs!Safety


rs.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End Function
Public Function GLBurden(OriginalAmt As Single, EmpID As Integer, WhichYear As Integer) As Single
Dim Yoda As DAO.Database, rs As DAO.Recordset, CPMax As Single, EIMax As Single
Dim Paidtodate As Single, CppPaid As Single, EIPaid As Single
Dim EPer As Single, CPer As Single
Dim VacPay As Single, CPPpay As Single, EIpay As Single
Set Yoda = CurrentDb

 
Set rs = Yoda.OpenRecordset("Select * from CPPEI where Yearly=" & WhichYear, dbOpenDynaset)
    If Not rs.EOF Then
        CPMax = rs!CPPMax
        EIMax = rs!EIMax
    End If
rs.Close

 
VacPay = CurrentBurden("vacpay")
CPPpay = CurrentBurden("Cpp")
EIpay = CurrentBurden("EI")

 
Set rs = Yoda.OpenRecordset("Select Sum(Amt) as Sumofamt From Payroll Where [Recorded by ID]=" & EmpID & " and Year([Ticket Date])=" & WhichYear)
    If Not rs.EOF Then
        Paidtodate = Round((Nz(rs!SumofAmt, 0) * (1 + VacPay)), 2)
        CppPaid = Paidtodate * CPPpay
        EIPaid = Paidtodate * EIpay
            'MsgBox "Total Paid " & Paidtodate
            'MsgBox "CPP Paid " & CppPaid
            'MsgBox "EI Paid " & EIPaid
        If CppPaid > CPMax Then CPer = 0 Else CPer = CPPpay
        If EIPaid > EIMax Then EPer = 0 Else EPer = EIpay
            'MsgBox CPer & vbCrLf & EPer
        GLBurden = Round(OriginalAmt * (1 + (CPer + EPer + VacPay)), 2)
    End If


 
rs.Close
Yoda.Close
Set Yoda = Nothing
Set rs = Nothing

End Function
