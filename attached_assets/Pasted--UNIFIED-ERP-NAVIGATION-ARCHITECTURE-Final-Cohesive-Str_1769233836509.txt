```sql
---------------------------------------------
-- NAVIGATION_ITEMS: Master Navigation Structure
---------------------------------------------
CREATE TABLE navigation_items (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    parent_id UUID REFERENCES navigation_items(id) ON DELETE CASCADE,
    
    -- Content & Structure
    item_order INTEGER NOT NULL DEFAULT 0,
    item_type TEXT NOT NULL CHECK (item_type IN ('menu', 'action', 'divider')),
    title TEXT NOT NULL,
    icon_name TEXT,
    path TEXT,
    component TEXT,
    
    -- 3/5 Rule Enforcement
    ui_slot TEXT DEFAULT 'sidebar' CHECK (ui_slot IN ('sidebar', 'topbar', 'dashboard')),
    max_children_display INTEGER DEFAULT 5 CHECK (max_children_display IN (3, 5)),
    
    -- Role-Based Access Control
    min_role_required TEXT DEFAULT 'viewer' 
        CHECK (min_role_required IN ('viewer', 'accountant', 'project_manager', 'admin')),
    
    -- Metadata
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    
    CONSTRAINT unique_tenant_path UNIQUE (tenant_id, path)
);

---------------------------------------------
-- ROLE_PERMISSIONS: Granular Access Control
---------------------------------------------
CREATE TABLE role_permissions (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    navigation_item_id UUID NOT NULL REFERENCES navigation_items(id) ON DELETE CASCADE,
    user_role TEXT NOT NULL CHECK (user_role IN ('viewer', 'accountant', 'project_manager', 'admin')),
    
    -- Permission Levels
    can_view BOOLEAN DEFAULT TRUE,
    can_access BOOLEAN DEFAULT TRUE,
    can_action BOOLEAN DEFAULT FALSE,
    
    -- Inheritance Control
    inherit_from_parent BOOLEAN DEFAULT TRUE,
    
    -- Contextual Permissions
    conditions JSONB DEFAULT '{}',
    
    UNIQUE(tenant_id, navigation_item_id, user_role)
);
```

## 🎯 **NAVIGATION SERVICE WITH 3/5 ENFORCEMENT**

### **Enhanced Navigation Service**
```javascript
// /lib/navigation-service.js
export class NavigationService {
    constructor(tenantId, userId, userRole) {
        this.tenantId = tenantId;
        this.userId = userId;
        this.userRole = userRole;
        this.supabase = createClient();
        this.roleHierarchy = { 'viewer': 0, 'accountant': 1, 'project_manager': 2, 'admin': 3 };
    }

    // MAIN METHOD: Get navigation with 3/5 enforcement
    async getNavigationForUser() {
        // 1. Get raw navigation tree
        const allItems = await this.fetchNavigationTree();
        
        // 2. Apply role-based filtering
        const filteredItems = this.filterByRoleAndPermissions(allItems);
        
        // 3. Enforce 3/5 choice constraint
        const constrainedTree = this.enforceChoiceConstraint(filteredItems);
        
        // 4. Apply user personalization (pinning, favorites)
        const personalizedTree = await this.applyPersonalization(constrainedTree);
        
        return personalizedTree;
    }

    // 3/5 CONSTRAINT ENFORCEMENT ENGINE
    enforceChoiceConstraint(items) {
        const processNode = (node) => {
            if (!node.children || node.children.length === 0) return node;
            
            const maxDisplay = node.max_children_display || 5;
            const children = [...node.children];
            
            // Sort: Pinned first, then by access frequency, then by order
            children.sort((a, b) => {
                if (a.is_pinned !== b.is_pinned) return a.is_pinned ? -1 : 1;
                if (a.access_count !== b.access_count) return b.access_count - a.access_count;
                return a.item_order - b.item_order;
            });
            
            // Apply limit
            let visibleChildren = children.slice(0, maxDisplay);
            const hiddenChildren = children.slice(maxDisplay);
            
            // Add "More..." action if items are hidden
            if (hiddenChildren.length > 0) {
                visibleChildren.push({
                    id: `more-${node.id}`,
                    title: `More (${hiddenChildren.length})`,
                    item_type: 'action',
                    icon_name: 'ChevronDown',
                    is_more_action: true,
                    hidden_children: hiddenChildren,
                    max_display: maxDisplay
                });
            }
            
            // Recurse on visible children
            node.children = visibleChildren.map(child => {
                if (!child.is_more_action) {
                    return processNode(child);
                }
                return child;
            });
            
            return node;
        };
        
        return items.map(processNode);
    }

    // ROLE-SPECIFIC FILTERING FOR PEOPLE SECTION
    filterPeopleSection(peopleNode, userRole) {
        const roleAccessMap = {
            'viewer': {
                visible: ['Contacts Directory'],
                canAccess: ['Contacts Directory']
            },
            'project_manager': {
                visible: ['Customers', 'Vendors & Pricing', 'Employees', 'Subcontractors', 'Contacts Directory'],
                canAccess: ['Customers', 'Vendors & Pricing', 'Subcontractors', 'Contacts Directory']
            },
            'accountant': {
                visible: ['Customers', 'Vendors & Pricing', 'Contacts Directory'],
                canAccess: ['Customers', 'Vendors & Pricing', 'Contacts Directory']
            },
            'admin': {
                visible: 'ALL',
                canAccess: 'ALL'
            }
        };
        
        const access = roleAccessMap[userRole] || roleAccessMap.viewer;
        
        if (access.visible === 'ALL') return peopleNode;
        
        peopleNode.children = peopleNode.children.filter(child => {
            const baseTitle = child.title.split('(')[0].trim();
            return access.visible.includes(baseTitle);
        });
        
        return peopleNode;
    }
}
```

```jsx
// /components/navigation/ERP_Sidebar.jsx
'use client';

import { NavigationService } from '@/lib/navigation-service';
import { PeopleSection } from './PeopleSection';
import { ChoiceTrio } from './ChoiceTrio';
import { ChoiceQuintet } from './ChoiceQuintet';

export function ERP_Sidebar({ tenantId, userId, userRole }) {
    const [navigation, setNavigation] = useState(null);
    const [activeSection, setActiveSection] = useState('dashboard');

    useEffect(() => {
        loadNavigation();
    }, [tenantId, userRole]);

    async function loadNavigation() {
        const service = new NavigationService(tenantId, userId, userRole);
        const nav = await service.getNavigationForUser();
        setNavigation(nav);
    }

    const renderSection = (section) => {
        const childCount = section.children?.length || 0;
        
        // Special handling for People section
        if (section.title === 'People & Contacts') {
            return (
                <PeopleSection 
                    key={section.id}
                    section={section}
                    userRole={userRole}
                    isActive={activeSection === 'people'}
                    onSelect={setActiveSection}
                />
            );
        }
        
        // Apply 3/5 rule based on child count
        if (childCount === 3) {
            return (
                <div key={section.id} className="nav-section">
                    <SectionHeader 
                        title={section.title} 
                        icon={section.icon_name}
                        isActive={activeSection === section.title.toLowerCase()}
                        onClick={() => setActiveSection(section.title.toLowerCase())}
                    />
                    {section.item_type === 'menu' && (
                        <ChoiceTrio
                            items={section.children}
                            onSelect={(item) => handleNavigation(item)}
                        />
                    )}
                </div>
            );
        } else if (childCount > 0 && childCount <= 5) {
            return (
                <div key={section.id} className="nav-section">
                    <SectionHeader 
                        title={section.title} 
                        icon={section.icon_name}
                        isActive={activeSection === section.title.toLowerCase()}
                        onClick={() => setActiveSection(section.title.toLowerCase())}
                    />
                    <ChoiceQuintet
                        items={section.children}
                        onSelect={(item) => handleNavigation(item)}
                        variant="list"
                    />
                </div>
            );
        }
        
        return null;
    };

    return (
        <aside className="erp-sidebar">
            {/* Logo & Tenant Selector */}
            <div className="sidebar-header">
                <TenantBranding tenantId={tenantId} />
            </div>
            
            {/* Main Navigation */}
            <nav className="sidebar-nav">
                {navigation?.map(renderSection)}
            </nav>
            
            {/* Fixed Bottom Actions (3 items) */}
            <div className="sidebar-footer">
                <ChoiceTrio
                    items={[
                        { id: 'help', title: 'Help', icon: 'HelpCircle', path: '/help' },
                        { id: 'settings', title: 'Settings', icon: 'Settings', path: '/settings' },
                        { id: 'profile', title: 'Profile', icon: 'User', path: '/profile' }
                    ]}
                    onSelect={handleNavigation}
                    compact={true}
                />
            </div>
        </aside>
    );
}
```

### **People & Contacts Section Component**
```jsx
// /components/navigation/PeopleSection.jsx
'use client';

export function PeopleSection({ section, userRole, isActive, onSelect }) {
    const [showMore, setShowMore] = useState(false);
    const [hiddenItems, setHiddenItems] = useState([]);
    
    const peopleItems = section.children || [];
    const visibleItems = peopleItems.filter(item => !item.is_more_action);
    const moreAction = peopleItems.find(item => item.is_more_action);
    
    const getItemStyle = (title) => {
        const styles = {
            'Customers': 'people-customer',
            'Vendors & Pricing': 'people-vendor',
            'Employees': 'people-employee',
            'Subcontractors': 'people-subcontractor',
            'Contacts Directory': 'people-directory'
        };
        return styles[title] || 'people-general';
    };
    
    const handleMoreClick = () => {
        if (moreAction?.hidden_children) {
            setHiddenItems(moreAction.hidden_children);
            setShowMore(!showMore);
        }
    };

    return (
        <div className={`people-section ${isActive ? 'active' : ''}`}>
            <SectionHeader 
                title="People & Contacts" 
                icon="Users"
                badge={visibleItems.length}
                isActive={isActive}
                onClick={() => onSelect('people')}
            />
            
            {/* 5-Item Grid Layout */}
            <div className="people-grid">
                {visibleItems.map(item => (
                    <PeopleCard
                        key={item.id}
                        item={item}
                        style={getItemStyle(item.title)}
                        onSelect={() => handleNavigation(item)}
                    />
                ))}
                
                {/* More Action Card */}
                {moreAction && (
                    <button 
                        className="people-card people-more"
                        onClick={handleMoreClick}
                        title={`Show ${moreAction.hidden_children?.length || 0} more items`}
                    >
                        <div className="people-icon">
                            <Icon name="ChevronDown" />
                        </div>
                        <div className="people-content">
                            <h4>{moreAction.title}</h4>
                            <p>Additional options</p>
                        </div>
                    </button>
                )}
            </div>
            
            {/* Hidden Items Dropdown */}
            {showMore && hiddenItems.length > 0 && (
                <div className="hidden-items-dropdown">
                    {hiddenItems.map(item => (
                        <PeopleCard
                            key={item.id}
                            item={item}
                            style={getItemStyle(item.title)}
                            onSelect={() => handleNavigation(item)}
                            compact={true}
                        />
                    ))}
                </div>
            )}
            
            {/* Role-Based Access Note */}
            {userRole === 'viewer' && (
                <div className="access-note">
                    <Icon name="Lock" size={14} />
                    <span>Upgrade role for full People access</span>
                </div>
            )}
        </div>
    );
}

// Individual People Card Component
function PeopleCard({ item, style, onSelect, compact = false }) {
    return (
        <button 
            className={`people-card ${style} ${compact ? 'compact' : ''}`}
            onClick={onSelect}
        >
            <div className="people-icon">
                <Icon name={item.icon_name} />
                {item.title === 'Employees' && (
                    <span className="sync-badge" title="BambooHR Synced">
                        <Icon name="RefreshCw" size={10} />
                    </span>
                )}
            </div>
            <div className="people-content">
                <h4>{item.title}</h4>
                {!compact && (
                    <p className="description">
                        {getPeopleDescription(item.title)}
                    </p>
                )}
            </div>
            {item.access_count > 0 && (
                <span className="access-count">{item.access_count}</span>
            )}
        </button>
    );
}
```
