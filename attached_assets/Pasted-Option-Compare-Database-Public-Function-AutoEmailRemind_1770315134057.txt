Public Sub AutoSendEmail(Optional emailto As String, Optional sentby As String, Optional CC As String, Optional Subject As String, Optional Body As String, Optional Att As Variant, Optional Att2 As Variant, Optional Att3 As Variant, Optional Att4 As Variant, Optional Att5 As Variant, Optional Att6 As Variant, Optional SendDirect As Boolean)
Dim ObjOutlook As Outlook.Application
Dim ObjEmail As Outlook.MailItem
Dim MyAttachments As Outlook.Attachments

        Set ObjOutlook = CreateObject("Outlook.application")
        Set ObjEmail = ObjOutlook.CreateItem(olMailItem)
        Set MyAttachments = ObjEmail.Attachments
            With ObjEmail
                .To = Nz(emailto, "Rob @ Salvi Homes <rob@salvi-homes.com>")
                .SentOnBehalfOfName = sentby
                .Subject = Subject
                .CC = sentby & ";" & CC
                .Body = Body & vbCrLf & vbCrLf & "This message is automatically generated by Salvi's - 'The Maestro'"
                If IsMissing(Att) = False Then MyAttachments.Add Att, olByValue, 1, ""
                If IsMissing(Att2) = False Then MyAttachments.Add Att2, olByValue, 1, ""
                If IsMissing(Att3) = False Then MyAttachments.Add Att3, olByValue, 1, ""
                If IsMissing(Att4) = False Then MyAttachments.Add Att4, olByValue, 1, ""
                If IsMissing(Att5) = False Then MyAttachments.Add Att5, olByValue, 1, ""
                If IsMissing(Att6) = False Then MyAttachments.Add Att6, olByValue, 1, ""
                If SendDirect <> True Then .Display Else .Send
            End With
        Set ObjEmail = Nothing
        Set ObjOutlook = Nothing

End Sub
