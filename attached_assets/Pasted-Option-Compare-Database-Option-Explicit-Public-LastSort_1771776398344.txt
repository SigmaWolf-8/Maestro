Public Sub PEItem_AfterUpdate()
Dim Yoda As DAO.Database, rs As DAO.Recordset, rsd As DAO.Recordset
Set Yoda = CurrentDb

Select Case Me.CostType.Column(1)
Case Is = "Material"
    'First Check to make Sure Vendor Exists
    Set rs = Yoda.OpenRecordset("Select * From dbo_Vendor Where Company='" & Me.Vendor & "'", dbOpenDynaset, dbSeeChanges)
    If rs.EOF Then
        MsgBox "This Vendor does not exist as a material supplier in the Pro Est Database" & vbCrLf & "Go to the Vendor form in The Maestro and add vendor first", vbOKOnly, "Add Vendor"
    Else
        Set rs = Yoda.OpenRecordset("Select * from dbo_ItemDetail where itemID=" & Nz(Me.PEItem, "1") & "", dbOpenDynaset, dbSeeChanges)
        Set rsd = Yoda.OpenRecordset("Select VendorID,Company from dbo_Vendor where Company='" & Me.Vendor & "'", dbOpenSnapshot)
If Not rs.EOF Then
        rs.Edit
        rs!VendorID = rsd!VendorID
        rs.Update
        'Update Dbo_ItemCostDetail Table
        If Me.DefVend = True Then
            Set rs = Yoda.OpenRecordset("Select * from Dbo_ItemCostDetail where itemID=" & Me.PEItem & " And CosttypeID=" & Me.CostType & "", dbOpenDynaset, dbSeeChanges)
            If Not rs.EOF Then
                rs.Edit
                rs!Cost = Me.Price
                rs.Update
            Else
                'rs.AddNew
                'rs.update
            End If
                
            'Update Dbo_ItemVendorDetail Table
            Set rs = Yoda.OpenRecordset("Select * from Dbo_ItemVendorDetail where itemID=" & Me.PEItem & " And VendorID=" & rsd!VendorID & "", dbOpenDynaset, dbSeeChanges)
            If Not rs.EOF Then
                rs.Edit
                rs!VendorID = rsd!VendorID
                rs!basecost = Me.Price
                rs!Cost = Me.Price
                rs!Partnumber = Me.SKU
                rs!editedby = 1
                rs!Editedon = Date
                rs!IsSelected = True
                rs.Update
            Else
                rs.AddNew
                rs!Itemid = Me.PEItem
                rs!VendorID = rsd!VendorID
                rs!basecost = Me.Price
                rs!Cost = Me.Price
                rs!Partnumber = Me.SKU
                rs!CreatedBy = 1
                rs!Createdon = Date
                rs!IsSelected = True
                rs.Update
            End If
            rsd.Close
            Set rsd = Nothing
        
        Else 'Vendor is not Default
            'Update Dbo_ItemVendorDetail Table
            Set rs = Yoda.OpenRecordset("Select * from Dbo_ItemVendorDetail where itemID=" & Me.PEItem & " And VendorID=" & rsd!VendorID & "", dbOpenDynaset, dbSeeChanges)
            
            If Not rs.EOF Then

                rs!Partnumber = Me.SKU
                rs!editedby = 1
                rs!Editedon = Date
                rs!IsSelected = False
                rs.Update
            Else
                rs.AddNew
                rs!Itemid = Me.PEItem
                rs!VendorID = rsd!VendorID
                rs!basecost = Me.Price
                rs!Cost = Me.Price
                rs!Partnumber = Me.SKU
                rs!CreatedBy = 1
                rs!Createdon = Date
                rs!IsSelected = False
                rs.Update
            End If
        
            rsd.Close
            Set rsd = Nothing
        End If
End If
    End If

rs.Close
Set rs = Nothing
Set rsd = Nothing



Case Is = "Labor"
