Private Sub Company_AfterUpdate()
Dim Yoda As DAO.Database, rs As DAO.Recordset, InsertNewVendor As Integer, UpdateVendor As Integer
Set Yoda = CurrentDb


'Me.Requery
Set Yoda = CurrentDb
Set rs = Yoda.OpenRecordset("Select * From SalviVendors Where Company='" & Me.Company & "'", dbOpenDynaset)

If rs.EOF Then
    If NewRecord = True Then
        InsertNewVendor = MsgBox("This Company does not exist in The Core database." & vbCrLf & "Would you like to add it?", vbYesNo, "Salvi Vendor Management")
        If InsertNewVendor = vbYes Then
            rs.AddNew
            rs!Company = Me.Company
            rs![Vendor ID] = Me.Company
            rs.Update
        End If
    Else
        UpdateVendor = MsgBox("Would you like to change this company spelling for all other listings under " & vbCrLf & Oldcompanyname, vbYesNo, "Salvi Vendor Management")
        If UpdateVendor = vbYes Then
            CurrentDb.Execute "UPDATE SalviVendors SET SalviVendors.Company = '" & Me.Company.Value & "' Where SalviVendors.Company='" & Oldcompanyname & "'"
            CurrentDb.Execute "UPDATE SalviContacts SET SalviContacts.Company = '" & Me.Company.Value & "' Where SalviContacts.Company='" & Oldcompanyname & "'"
        End If
    End If
End If

Yoda.Close
Set Yoda = Nothing
Set rs = Nothing
End Sub
